// Zephra, the Windborne - evasive arcane-trickster gish-rogue.
// Glass cannon built around one lethal opening: she moves first, closes on whoever
// has not acted yet, and spends the whole round killing that one target.

USING rogue_smart

// The opening is the kill, not the buff. Illusion and invisibility come after.
MULTIPLIER_FIRST_ACTION_BUFF                    1.00
MULTIPLIER_FIRST_ACTION_INVISIBILITY            0.10
MULTIPLIER_INVISIBLE                            0.15

// Her strike is melee, so the end of the move has to sit inside reach. The max distance is
// where the gradient stops: past it every spot scores the same, and a teleport picking from
// a wide ring of equal-scoring spots lands anywhere. It has to span the whole hop.
MULTIPLIER_ENDPOS_ENEMIES_NEARBY                0.20
ENDPOS_ENEMIES_NEARBY_MAX_DISTANCE              9.00
ENABLE_MOVEMENT_AVOID_AOO                       1.00

// Wind Step is an approach, not an exit. The default 6m floor rules out the short
// hop onto a target standing nearby, and the strike planned behind the teleport is
// what the landing spot is chosen for. Leaving a position that already works keeps
// the full vanilla penalty.
SPELL_TELEPORT_MINIMUM_DISTANCE                 3.00
MULTIPLIER_PLANNED_ACTION_WITH_MOVE_SPELL       2.50
MULTIPLIER_POSITION_LEAVE                      -1.00

// Picks the one she can drop this round and commits to it.
MULTIPLIER_KILL_ENEMY                           2.00
MULTIPLIER_TARGET_HEALTH_BIAS                   0.25
MULTIPLIER_KILL_TARGET_HEALTH_BIAS              0.20
MULTIPLIER_DAMAGE_ENEMY_POS                     1.35

// Field control is a side tool, kept for the half of the fight she survives into.
MULTIPLIER_CONTROL_ENEMY_POS                    0.80
MULTIPLIER_BOOST_ENEMY_POS                      0.90
MODIFIER_CONCENTRATION_REMOVE_TARGET            0.40

// Glass cannon: protective of her own skin
MULTIPLIER_DAMAGE_SELF_NEG                      3.00

// A precise duelist: drinks potions, but never throws
USE_INVENTORY_ITEMS_ENABLED                     1
THROW_INVENTORY_ITEM_LIMIT                       0
MULTIPLIER_USABLE_ITEM                          0.00
WEAPON_PICKUP_MODIFIER                          0.00

// Her own sleet lies under every fight: crossing it costs her nothing, and only the tile
// she lands on is weighed.
MULTIPLIER_MOVEMENT_SURFACE                     0.00
MULTIPLIER_ENDPOS_NOT_IN_DANGEROUS_SURFACE      0.15
