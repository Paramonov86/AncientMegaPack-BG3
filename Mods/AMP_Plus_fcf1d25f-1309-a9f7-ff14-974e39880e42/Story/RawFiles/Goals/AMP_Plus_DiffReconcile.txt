SubGoalCombiner SGC_AND
INITSECTION

KBSECTION

// base never re-grades on a mode switch

IF
DB_AMP_Difficulty_Mode((STRING)_Mode)
AND
_Mode != "NONE"
AND
GetHostCharacter((CHARACTER)_Host)
AND
NOT IsInCombat(_Host, 1)
THEN
PROC_AMP_Plus_DiffReconcile();

// defer: a live re-grade re-applies max hp and heals enemies
IF
DB_AMP_Difficulty_Mode((STRING)_Mode)
AND
_Mode != "NONE"
AND
GetHostCharacter((CHARACTER)_Host)
AND
IsInCombat(_Host, 1)
THEN
DB_AMP_Plus_DiffReconcilePending(1);

IF
DB_AMP_Difficulty_Mode("NONE")
THEN
PROC_AMP_DiffV3_RetierAll_Collect();
PROC_AMP_DiffV3_RetierAll_Strip();
PROC_AMP_Plus_Feral_ClearBumpMarkers();

IF
CombatEnded((GUIDSTRING)_C)
AND
DB_AMP_Plus_DiffReconcilePending(1)
THEN
NOT DB_AMP_Plus_DiffReconcilePending(1);
PROC_AMP_Plus_DiffReconcile();

IF
LevelGameplayStarted(_RecLvl, _RecEditor)
AND
DB_AMP_Plus_DiffReconcilePending(1)
THEN
NOT DB_AMP_Plus_DiffReconcilePending(1);

PROC
PROC_AMP_Plus_DiffReconcile()
THEN
PROC_AMP_UpdateAvgPartyLevel();
PROC_AMP_UpdateCurrentTier();
PROC_AMP_DiffV3_RetierAll_Collect();
PROC_AMP_DiffV3_RetierAll_Strip();
PROC_AMP_Plus_Feral_ClearBumpMarkers();
PROC_AMP_DiffV3_RetierAll_Reapply();

EXITSECTION

ENDEXITSECTION
