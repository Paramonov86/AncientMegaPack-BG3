_Idx <= _Total
AND
CombatGetInvolvedPartyMember(_CombatGuid, _Idx, (CHARACTER)_Member)
AND
DB_PartOfTheTeam(_Member)
THEN
DB_AMP_Plus_PChest_InFight(_CombatGuid, _Member);
DB_AMP_Plus_PChest_TeamFight(_CombatGuid);

PROC
PROC_AMP_Plus_PChest_Mark((GUIDSTRING)_CombatGuid, (INTEGER)_Total, (INTEGER)_Idx)
AND
_Idx <= _Total
AND
IntegerSum(_Idx, 1, (INTEGER)_Next)
THEN
PROC_AMP_Plus_PChest_Mark(_CombatGuid, _Total, _Next);

AND
