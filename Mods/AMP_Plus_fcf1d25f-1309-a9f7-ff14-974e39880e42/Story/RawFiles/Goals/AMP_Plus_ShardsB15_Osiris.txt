PROC
PROC_AMP_Plus_ShardRecurrence_Seed()
AND
DB_Players((CHARACTER)_Char)
AND
DB_AMP_Plus_ShardRecurrence_Toll((STRING)_Boon, (STRING)_Old)
AND
HasActiveStatus(_Char, _Old, 1)
THEN
RemoveStatus(_Char, _Old, _Char);
DB_AMP_Plus_ShardRecurrence_Charged(_Char, _Boon, "AMP_SHARD_RECURRENCE_TOLL_HP");
PROC_AMP_Plus_ShardRecurrence_Rebuild(_Char);

PROC
PROC_AMP_Plus_ShardRecurrence_DrainWatch()
AND
DB_AMP_Plus_ShardRecurrence_Watch((STRING)_Status)
THEN
NOT DB_AMP_Plus_ShardRecurrence_Watch(_Status);
PROC_AMP_Plus_ShardRecurrence_DrainWatch();

PROC
PROC_AMP_Plus_ShardRecurrence_DrainToll()
AND
DB_AMP_Plus_ShardRecurrence_Toll((STRING)_Status, (STRING)_Toll)
THEN
NOT DB_AMP_Plus_ShardRecurrence_Toll(_Status, _Toll);
PROC_AMP_Plus_ShardRecurrence_DrainToll();

// the pick lives on the helmet status, character statuses are rebuilt on every load

IF
StatusApplied((ITEM)_Item, "AMP_SHARD_HEAD_RECURRENCE", (CHARACTER)_Causee, _)
AND
DB_Players(_Causee)
AND
IsEquipped(_Item, 1)
AND
GetInventoryOwner(_Item, (CHARACTER)_Char)
THEN
PROC_AMP_Plus_ShardRecurrence_ForgetPick(_Item);
PROC_AMP_Plus_ShardRecurrence_OfferPick(_Char);

IF
StatusRemoved((ITEM)_Item, "AMP_SHARD_HEAD_RECURRENCE", _, _)
THEN
PROC_AMP_Plus_ShardRecurrence_ForgetPick(_Item);

IF
StatusRemoved((ITEM)_Item, "AMP_SHARD_HEAD_RECURRENCE", _, _)
AND
GetInventoryOwner(_Item, (CHARACTER)_Char)
AND
DB_AMP_Plus_ShardRecurrence_ChoiceOpen(_Char)
THEN
NOT DB_AMP_Plus_ShardRecurrence_ChoiceOpen(_Char);
RemoveSpell(_Char, "Shout_AMP_Plus_Recurrence_TollChoice", 1);

IF
Equipped((ITEM)_Item, (CHARACTER)_Char)
AND
HasActiveStatus(_Item, "AMP_SHARD_HEAD_RECURRENCE", 1)
AND
NOT DB_AMP_Plus_ShardRecurrence_Pick(_Item, _)
THEN
PROC_AMP_Plus_ShardRecurrence_OfferPick(_Char);

IF
SavegameLoaded()
AND
DB_Players((CHARACTER)_Char)
AND
GetEquippedItem(_Char, "Helmet", (ITEM)_Item)
AND
HasActiveStatus(_Item, "AMP_SHARD_HEAD_RECURRENCE", 1)
AND
NOT DB_AMP_Plus_ShardRecurrence_Pick(_Item, _)
THEN
PROC_AMP_Plus_ShardRecurrence_OfferPick(_Char);

PROC
PROC_AMP_Plus_ShardRecurrence_ForgetPick((ITEM)_Item)
AND
DB_AMP_Plus_ShardRecurrence_Pick(_Item, (STRING)_Toll)
THEN
NOT DB_AMP_Plus_ShardRecurrence_Pick(_Item, _Toll);

PROC
PROC_AMP_Plus_ShardRecurrence_OfferPick((CHARACTER)_Char)
AND
NOT DB_AMP_Plus_ShardRecurrence_ChoiceOpen(_Char)
THEN
DB_AMP_Plus_ShardRecurrence_ChoiceOpen(_Char);
AddSpell(_Char, "Shout_AMP_Plus_Recurrence_TollChoice", 0, 1);

IF
UsingSpell((CHARACTER)_Char, "Shout_AMP_Plus_Recurrence_Pick_HP", _, _, _)
THEN
PROC_AMP_Plus_ShardRecurrence_TakePick(_Char, "AMP_SHARD_RECURRENCE_TOLL_HP");

IF
UsingSpell((CHARACTER)_Char, "Shout_AMP_Plus_Recurrence_Pick_AC", _, _, _)
THEN
PROC_AMP_Plus_ShardRecurrence_TakePick(_Char, "AMP_SHARD_RECURRENCE_TOLL_AC");

// removing the spell mid-shout cancels the cast, withdraw after the animation

PROC
PROC_AMP_Plus_ShardRecurrence_TakePick((CHARACTER)_Char, (STRING)_Toll)
AND
GetEquippedItem(_Char, "Helmet", (ITEM)_Item)
AND
HasActiveStatus(_Item, "AMP_SHARD_HEAD_RECURRENCE", 1)
AND
NOT DB_AMP_Plus_ShardRecurrence_Pick(_Item, _)
THEN
DB_AMP_Plus_ShardRecurrence_Pick(_Item, _Toll);
ObjectTimerLaunch(_Char, "AMP_Plus_ShardRecurrence_ClosePick", 3000);
PROC_AMP_Plus_ShardRecurrence_Switch(_Char, _Toll);
PROC_AMP_Plus_ShardRecurrence_Rebuild(_Char);

PROC
PROC_AMP_Plus_ShardRecurrence_Switch((CHARACTER)_Char, (STRING)_Toll)
AND
DB_AMP_Plus_ShardRecurrence_Charged(_Char, (STRING)_Boon, (STRING)_Old)
AND
_Old != _Toll
THEN
NOT DB_AMP_Plus_ShardRecurrence_Charged(_Char, _Boon, _Old);
DB_AMP_Plus_ShardRecurrence_Charged(_Char, _Boon, _Toll);
PROC_AMP_Plus_ShardRecurrence_Switch(_Char, _Toll);

IF
ObjectTimerFinished((CHARACTER)_Char, "AMP_Plus_ShardRecurrence_ClosePick")
THEN
NOT DB_AMP_Plus_ShardRecurrence_ChoiceOpen(_Char);
RemoveSpell(_Char, "Shout_AMP_Plus_Recurrence_TollChoice", 1);

// rest strips these boons before it ends, so read at start and lay back at end

IF
LongRestStarted()
AND
DB_Players((CHARACTER)_Char)
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_RECURRENCE", 1)
AND
DB_AMP_Plus_ShardRecurrence_Watch((STRING)_Status)
AND
HasActiveStatus(_Char, _Status, 1)
THEN
DB_AMP_Plus_ShardRecurrence_Kept(_Char, _Status);

IF
LongRestFinished()
THEN
PROC_AMP_Plus_ShardRecurrence_DrainKeptReturn();

PROC
PROC_AMP_Plus_ShardRecurrence_DrainKeptReturn()
AND
DB_AMP_Plus_ShardRecurrence_Kept((CHARACTER)_Char, (STRING)_Status)
THEN
NOT DB_AMP_Plus_ShardRecurrence_Kept(_Char, _Status);
PROC_AMP_Plus_ShardRecurrence_Return(_Char, _Status);
PROC_AMP_Plus_ShardRecurrence_DrainKeptReturn();

PROC
PROC_AMP_Plus_ShardRecurrence_Return((CHARACTER)_Char, (STRING)_Status)
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_RECURRENCE", 1)
AND
HasActiveStatus(_Char, _Status, 0)
THEN
ApplyStatus(_Char, _Status, -1.0, 1, _Char);
PROC_AMP_Plus_ShardRecurrence_Charge(_Char, _Status);
PROC_AMP_Plus_ShardRecurrence_Rebuild(_Char);

// a region change is a rest without the rest: the buffs are gone before anything can read them,
// so every watched buff seen on the wearer is remembered and laid back on the next level

IF
StatusApplied((CHARACTER)_Char, (STRING)_Status, _, _)
AND
DB_Players(_Char)
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_RECURRENCE", 1)
AND
DB_AMP_Plus_ShardRecurrence_Watch(_Status)
THEN
DB_AMP_Plus_ShardRecurrence_Held(_Char, _Status);

IF
StatusRemoved((CHARACTER)_Char, "AMP_SHARD_CHAR_RECURRENCE", _, _)
THEN
PROC_AMP_Plus_ShardRecurrence_DrainHeld(_Char);

PROC
PROC_AMP_Plus_ShardRecurrence_DrainHeld((CHARACTER)_Char)
AND
DB_AMP_Plus_ShardRecurrence_Held(_Char, (STRING)_Status)
THEN
NOT DB_AMP_Plus_ShardRecurrence_Held(_Char, _Status);
PROC_AMP_Plus_ShardRecurrence_DrainHeld(_Char);

IF
LevelGameplayStarted(_, _)
AND
DB_Players((CHARACTER)_Char)
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_RECURRENCE", 1)
AND
DB_AMP_Plus_ShardRecurrence_Held(_Char, (STRING)_Status)
AND
NOT HasActiveStatus(_Char, _Status, 1)
THEN
PROC_AMP_Plus_ShardRecurrence_Return(_Char, _Status);

PROC
PROC_AMP_Plus_ShardRecurrence_Charge((CHARACTER)_Char, (STRING)_Status)
AND
GetEquippedItem(_Char, "Helmet", (ITEM)_Item)
AND
DB_AMP_Plus_ShardRecurrence_Pick(_Item, (STRING)_Toll)
THEN
DB_AMP_Plus_ShardRecurrence_Charged(_Char, _Status, _Toll);

PROC
PROC_AMP_Plus_ShardRecurrence_Charge((CHARACTER)_Char, (STRING)_Status)
AND
NOT DB_AMP_Plus_ShardRecurrence_Charged(_Char, _Status, _)
THEN
DB_AMP_Plus_ShardRecurrence_Charged(_Char, _Status, "AMP_SHARD_RECURRENCE_TOLL_HP");

// no StackId: strip all, lay one back per boon, a second apart or AC keeps the removal

IF
StatusRemoved((CHARACTER)_Char, (STRING)_Status, _, _)
AND
DB_AMP_Plus_ShardRecurrence_Charged(_Char, _Status, (STRING)_Toll)
THEN
NOT DB_AMP_Plus_ShardRecurrence_Charged(_Char, _Status, _Toll);
PROC_AMP_Plus_ShardRecurrence_Rebuild(_Char);

PROC
PROC_AMP_Plus_ShardRecurrence_Rebuild((CHARACTER)_Char)
THEN
RemoveStatus(_Char, "AMP_SHARD_RECURRENCE_TOLL_HP", NULL_00000000-0000-0000-0000-000000000000);
RemoveStatus(_Char, "AMP_SHARD_RECURRENCE_TOLL_AC", NULL_00000000-0000-0000-0000-000000000000);
DB_AMP_Plus_ShardRecurrence_Rebuilding(_Char);
ObjectTimerLaunch(_Char, "AMP_Plus_ShardRecurrence_Rebuild", 1000);

IF
ObjectTimerFinished((CHARACTER)_Char, "AMP_Plus_ShardRecurrence_Rebuild")
AND
DB_AMP_Plus_ShardRecurrence_Rebuilding(_Char)
THEN
NOT DB_AMP_Plus_ShardRecurrence_Rebuilding(_Char);
PROC_AMP_Plus_ShardRecurrence_LayTolls(_Char);

PROC
PROC_AMP_Plus_ShardRecurrence_LayTolls((CHARACTER)_Char)
AND
DB_AMP_Plus_ShardRecurrence_Charged(_Char, _, (STRING)_Toll)
THEN
ApplyStatus(_Char, _Toll, -1.0, 1, _Char);

IF
LongRestCancelled()
THEN
PROC_AMP_Plus_ShardRecurrence_DrainKept();

PROC
PROC_AMP_Plus_ShardRecurrence_DrainKept()
AND
DB_AMP_Plus_ShardRecurrence_Kept((CHARACTER)_Char, (STRING)_Status)
THEN
NOT DB_AMP_Plus_ShardRecurrence_Kept(_Char, _Status);
PROC_AMP_Plus_ShardRecurrence_DrainKept();

// UND_BOOOALBLESSING has no StackId and ignores rest: strip, hand back one

IF
LongRestFinished()
AND
DB_Players((CHARACTER)_Char)
AND
NOT DB_AMP_Plus_ShardRecurrence_Destacked(_Char)
AND
HasActiveStatus(_Char, "UND_BOOOALBLESSING", 1)
THEN
DB_AMP_Plus_ShardRecurrence_Destacked(_Char);
PROC_AMP_Plus_ShardRecurrence_Destack(_Char, 12);

PROC
PROC_AMP_Plus_ShardRecurrence_Destack((CHARACTER)_Char, (INTEGER)_Left)
AND
_Left > 0
AND
IntegerSubtract(_Left, 1, (INTEGER)_Next)
THEN
RemoveStatus(_Char, "UND_BOOOALBLESSING", NULL_00000000-0000-0000-0000-000000000000);
PROC_AMP_Plus_ShardRecurrence_Destack(_Char, _Next);

PROC
PROC_AMP_Plus_ShardRecurrence_Destack((CHARACTER)_Char, (INTEGER)_Left)
AND
_Left <= 0
THEN
ApplyStatus(_Char, "UND_BOOOALBLESSING", -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);


// Trepanation reads a marker, not Ally(): a maddened wearer has no allies left

IF
EnteredCombat((CHARACTER)_Char, (GUIDSTRING)_CombatID)
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_TREPANATION", 1)
AND
CombatGetInvolvedPartyMembersCount(_CombatID, (INTEGER)_N)
AND
_N > 0
THEN
PROC_AMP_Plus_ShardTrepanation_Mark(_Char, _CombatID, _N, 1);

IF
EnteredCombat((CHARACTER)_Char, (GUIDSTRING)_CombatID)
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_TREPANATION", 1)
AND
DB_Players((CHARACTER)_Player)
THEN
ApplyStatus(_Player, "AMP_SHARD_TREPANATION_FRIEND", -1.0, 1, _Char);

PROC
PROC_AMP_Plus_ShardTrepanation_Mark((CHARACTER)_Char, (GUIDSTRING)_CombatID, (INTEGER)_Total, (INTEGER)_Idx)
AND
_Idx <= _Total
AND
CombatGetInvolvedPartyMember(_CombatID, _Idx, (CHARACTER)_Member)
AND
IsEnemy(_Member, _Char, 0)
AND
IntegerSum(_Idx, 1, (INTEGER)_Next)
THEN
ApplyStatus(_Member, "AMP_SHARD_TREPANATION_FRIEND", -1.0, 1, _Char);
PROC_AMP_Plus_ShardTrepanation_Mark(_Char, _CombatID, _Total, _Next);

PROC
PROC_AMP_Plus_ShardTrepanation_Mark((CHARACTER)_Char, (GUIDSTRING)_CombatID, (INTEGER)_Total, (INTEGER)_Idx)
AND
_Idx <= _Total
AND
CombatGetInvolvedPartyMember(_CombatID, _Idx, (CHARACTER)_Member)
AND
IsEnemy(_Member, _Char, 1)
AND
IntegerSum(_Idx, 1, (INTEGER)_Next)
THEN
PROC_AMP_Plus_ShardTrepanation_Mark(_Char, _CombatID, _Total, _Next);

PROC

AND
IntegerSum(_Old, 1, (INTEGER)_New)
THEN
NOT DB_AMP_Plus_Lamb_Gifts(_W, _Marker, _Old);
DB_AMP_Plus_Lamb_Gifts(_W, _Marker, _New);

// keep in step with DescriptionParams [2] of the Lamb shards
PROC
PROC_AMP_Plus_Lamb_Give((CHARACTER)_Char, (ITEM)_W, (STRING)_Marker, (ITEMROOT)_Capsule)

DB_AMP_Plus_Lamb_Gifts(_W, _Marker, (INTEGER)_N)
AND
_N >= 5
THEN

PROC_AMP_Plus_Lamb_Burn((CHARACTER)_Char, (ITEM)_W, (STRING)_Marker)
AND
DB_AMP_Plus_Lamb_Gifts(_W, _Marker, (INTEGER)_N)
THEN
NOT DB_AMP_Plus_Lamb_Gifts(_W, _Marker, _N);

// counter row goes before the marker, so the removal frees no slot on its own

PROC_AMP_Plus_Lamb_Burn((CHARACTER)_Char, (ITEM)_W, (STRING)_Marker)
AND
DB_AMP_Shard_Count(_W, _Marker, (INTEGER)_Count)
THEN
NOT DB_AMP_Shard_Count(_W, _Marker, _Count);

// the re-equip backup would otherwise put the marker back
PROC
PROC_AMP_Plus_Lamb_Burn((CHARACTER)_Char, (ITEM)_W, (STRING)_Marker)
AND
DB_AMP_Shard_Saved(_W, _Marker, (INTEGER)_Count)
THEN
NOT DB_AMP_Shard_Saved(_W, _Marker, _Count);

PROC_AMP_Plus_Lamb_Burn((CHARACTER)_Char, (ITEM)_W, (STRING)_Marker)
THEN
PROC_AMP_Shard_DecTotal(_W);
RemoveStatus(_W, _Marker, _Char);
PROC_AMP_RecoverTransferCleanup(_Char, _Marker);
ObjectTimerLaunch(_Char, "AMP_Plus_Lamb_Resync", 500);

// lamb statuses on the wearer share one StackId, so another lamb weapon may have lost its own
IF
ObjectTimerFinished((CHARACTER)_Char, "AMP_Plus_Lamb_Resync")
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_SACRAMENT", 0)
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_OBLATION", 0)
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_HECATOMB", 0)
AND
HasActiveStatus(_Char, "AMP_SHARD_CHAR_CONSUMMATION", 0)
THEN
PROC_AMP_Plus_Lamb_ResyncSlot(_Char, "Melee Main Weapon");
PROC_AMP_Plus_Lamb_ResyncSlot(_Char, "Ranged Main Weapon");
PROC_AMP_Plus_Lamb_ResyncSlot(_Char, "Melee Offhand Weapon");
PROC_AMP_Plus_Lamb_ResyncSlot(_Char, "Ranged Offhand Weapon");
PROC_AMP_Plus_Lamb_ResyncDone(_Char);

PROC
PROC_AMP_Plus_Lamb_ResyncSlot((CHARACTER)_Char, (STRING)_Slot)
AND
GetEquippedItem(_Char, _Slot, (ITEM)_W)
THEN
PROC_AMP_Plus_Lamb_ResyncItem(_Char, _W, "AMP_SHARD_WEAPON_SACRAMENT");
PROC_AMP_Plus_Lamb_ResyncItem(_Char, _W, "AMP_SHARD_WEAPON_OBLATION");
PROC_AMP_Plus_Lamb_ResyncItem(_Char, _W, "AMP_SHARD_WEAPON_HECATOMB");
PROC_AMP_Plus_Lamb_ResyncItem(_Char, _W, "AMP_SHARD_WEAPON_CONSUMMATION");

PROC
PROC_AMP_Plus_Lamb_ResyncItem((CHARACTER)_Char, (ITEM)_W, (STRING)_Marker)
AND
HasActiveStatus(_W, _Marker, 1)
AND
NOT DB_AMP_Plus_Lamb_Resynced(_Char)
AND
DB_AMP_Shard_Transfer(_Marker, (STRING)_CharStatus)
THEN
DB_AMP_Plus_Lamb_Resynced(_Char);
ApplyStatus(_Char, _CharStatus, -1.0, 1, _Char);

PROC
PROC_AMP_Plus_Lamb_ResyncDone((CHARACTER)_Char)
AND
DB_AMP_Plus_Lamb_Resynced(_Char)
THEN
NOT DB_AMP_Plus_Lamb_Resynced(_Char);

// a row left on a weapon that no longer carries the marker is not the shard being inserted now
IF
TemplateUseStarted((CHARACTER)_Char, (ITEMROOT)_Template, (ITEM)_Instance)
AND
DB_AMP_Shard_Weapon_NonStack(_Template, _, _, _Marker)
AND
HasMeleeWeaponEquipped(_Char, "Mainhand", 1)
AND
GetEquippedItem(_Char, "Melee Main Weapon", (ITEM)_W)
AND
DB_AMP_Plus_Lamb_Gifts(_W, _Marker, (INTEGER)_N)
AND
HasActiveStatus(_W, _Marker, 0)
THEN
NOT DB_AMP_Plus_Lamb_Gifts(_W, _Marker, _N);

IF
TemplateUseStarted((CHARACTER)_Char, (ITEMROOT)_Template, (ITEM)_Instance)
AND
DB_AMP_Shard_Weapon_NonStack(_Template, _, _, _Marker)
AND
HasMeleeWeaponEquipped(_Char, "Mainhand", 0)
AND
HasRangedWeaponEquipped(_Char, "Mainhand", 1)
AND
GetEquippedItem(_Char, "Ranged Main Weapon", (ITEM)_W)
AND
DB_AMP_Plus_Lamb_Gifts(_W, _Marker, (INTEGER)_N)
AND
HasActiveStatus(_W, _Marker, 0)
THEN
NOT DB_AMP_Plus_Lamb_Gifts(_W, _Marker, _N);
