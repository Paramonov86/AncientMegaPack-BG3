AND

AND

DB_AMP_Plus_RandState(_Boss, _Eff, _Size, _K2, _BX, _BY, _BZ);

// enabled-only picker: a full-pool roll loses a slot per disabled entry it lands on
// bag: a picked entry leaves, a batch larger than the pool refills it

PROC
PROC_AMP_Plus_MirrorRandSeedB((CHARACTER)_Boss, (INTEGER)_Eff, (INTEGER)_Size, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
THEN
PROC_AMP_Plus_EnPoolClear(_Boss);
DB_AMP_Plus_EnCount(_Boss, 0);
PROC_AMP_Plus_EnPoolBuild(_Boss, 0, _Size);
PROC_AMP_Plus_MirrorRandStartB(_Boss, _Eff, _Size, _BX, _BY, _BZ);

PROC
PROC_AMP_Plus_EnPoolBuild((CHARACTER)_Boss, (INTEGER)_I, (INTEGER)_Size)
AND
_I < _Size
AND
NOT DB_AMP_MirrorBoss_Disabled(_I)
AND
DB_AMP_MirrorPool(_I, _, _)
AND
DB_AMP_Plus_EnCount(_Boss, (INTEGER)_EC)
AND
IntegerSum(_EC, 1, (INTEGER)_EC2)
AND
IntegerSum(_I, 1, (INTEGER)_I2)
THEN
DB_AMP_Plus_EnPool(_Boss, _EC, _I);
NOT DB_AMP_Plus_EnCount(_Boss, _EC);
DB_AMP_Plus_EnCount(_Boss, _EC2);
PROC_AMP_Plus_EnPoolBuild(_Boss, _I2, _Size);

PROC
PROC_AMP_Plus_EnPoolBuild((CHARACTER)_Boss, (INTEGER)_I, (INTEGER)_Size)
AND
_I < _Size
AND
DB_AMP_MirrorBoss_Disabled(_I)
AND
IntegerSum(_I, 1, (INTEGER)_I2)
THEN
PROC_AMP_Plus_EnPoolBuild(_Boss, _I2, _Size);

// slot with no pool entry is skipped so the walk cannot stall
PROC
PROC_AMP_Plus_EnPoolBuild((CHARACTER)_Boss, (INTEGER)_I, (INTEGER)_Size)
AND
_I < _Size
AND
NOT DB_AMP_MirrorBoss_Disabled(_I)
AND
NOT DB_AMP_MirrorPool(_I, _, _)
AND
IntegerSum(_I, 1, (INTEGER)_I2)
THEN
PROC_AMP_Plus_EnPoolBuild(_Boss, _I2, _Size);

PROC
PROC_AMP_Plus_EnPoolBuild((CHARACTER)_Boss, (INTEGER)_I, (INTEGER)_Size)
AND
_I >= _Size
THEN
DB_NOOP(1);

// state carries the pool size, the list shrinks with every pick
PROC
PROC_AMP_Plus_MirrorRandStartB((CHARACTER)_Boss, (INTEGER)_Eff, (INTEGER)_Size, (REAL)_BX, (REAL)_BY, (REAL)_BZ)

_EC > 0
THEN
DB_AMP_Plus_RandStateB(_Boss, _Eff, _Size, 0, _BX, _BY, _BZ);

PROC
PROC_AMP_Plus_MirrorRandStartB((CHARACTER)_Boss, (INTEGER)_Eff, (INTEGER)_Size, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
DB_AMP_Plus_EnCount(_Boss, (INTEGER)_EC)
AND
_EC <= 0
THEN
PROC_AMP_Plus_EnPoolClear(_Boss);

PROC
PROC_AMP_Plus_MirrorRandStepB((CHARACTER)_Boss)
AND
DB_AMP_Plus_RandStateB(_Boss, (INTEGER)_Eff, (INTEGER)_Size, (INTEGER)_K, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_K < _Eff
AND
DB_AMP_Plus_EnCount(_Boss, (INTEGER)_EC)
AND
_EC > 0
AND
Random(_EC, (INTEGER)_J)
AND
DB_AMP_Plus_EnPool(_Boss, _J, (INTEGER)_Idx)
AND
DB_AMP_MirrorPool(_Idx, (GUIDSTRING)_Template, _)
AND
IntegerSum(_K, 1, (INTEGER)_K2)
THEN
PROC_AMP_Plus_SpawnAtTarget(_Boss, _Template, _K, _BX, _BY, _BZ);
NOT DB_AMP_Plus_RandStateB(_Boss, _Eff, _Size, _K, _BX, _BY, _BZ);
DB_AMP_Plus_RandStateB(_Boss, _Eff, _Size, _K2, _BX, _BY, _BZ);
PROC_AMP_Plus_BagTake(_Boss, _J);
PROC_AMP_Plus_MirrorRandStepB(_Boss);

PROC
PROC_AMP_Plus_MirrorRandStepB((CHARACTER)_Boss)
AND
DB_AMP_Plus_RandStateB(_Boss, (INTEGER)_Eff, (INTEGER)_Size, (INTEGER)_K, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_K < _Eff
AND
DB_AMP_Plus_EnCount(_Boss, (INTEGER)_EC)
AND
_EC <= 0
THEN

PROC
PROC_AMP_Plus_MirrorRandStepB((CHARACTER)_Boss)
AND
DB_AMP_Plus_RandStateB(_Boss, (INTEGER)_Eff, (INTEGER)_Size, (INTEGER)_K, (REAL)_BX, (REAL)_BY, (REAL)_BZ)

_K >= _Eff
THEN
NOT DB_AMP_Plus_RandStateB(_Boss, _Eff, _Size, _K, _BX, _BY, _BZ);
PROC_AMP_Plus_EnPoolClear(_Boss);

// three ordered rules, the count falls only in the last one

PROC
PROC_AMP_Plus_BagTake((CHARACTER)_Boss, (INTEGER)_J)
AND
DB_AMP_Plus_EnCount(_Boss, (INTEGER)_EC)
AND
IntegerSubtract(_EC, 1, (INTEGER)_Last)
AND
NOT _J == _Last
AND
DB_AMP_Plus_EnPool(_Boss, _J, (INTEGER)_IdxJ)
AND
DB_AMP_Plus_EnPool(_Boss, _Last, (INTEGER)_IdxLast)
THEN
NOT DB_AMP_Plus_EnPool(_Boss, _J, _IdxJ);
NOT DB_AMP_Plus_EnPool(_Boss, _Last, _IdxLast);
DB_AMP_Plus_EnPool(_Boss, _J, _IdxLast);

PROC
PROC_AMP_Plus_BagTake((CHARACTER)_Boss, (INTEGER)_J)
AND
DB_AMP_Plus_EnCount(_Boss, (INTEGER)_EC)
AND
IntegerSubtract(_EC, 1, (INTEGER)_Last)
AND
_J == _Last
AND
DB_AMP_Plus_EnPool(_Boss, _J, (INTEGER)_IdxJ)
THEN
NOT DB_AMP_Plus_EnPool(_Boss, _J, _IdxJ);

PROC
PROC_AMP_Plus_BagTake((CHARACTER)_Boss, (INTEGER)_J)
AND
DB_AMP_Plus_EnCount(_Boss, (INTEGER)_EC)
AND
IntegerSubtract(_EC, 1, (INTEGER)_EC2)
THEN
NOT DB_AMP_Plus_EnCount(_Boss, _EC);
DB_AMP_Plus_EnCount(_Boss, _EC2);
