// non-stackable markers, no counter row
PROC
PROC_AMP_Plus_RecoverFromSlot((CHARACTER)_Char, (STRING)_Slot)
AND
GetEquippedItem(_Char, _Slot, _Item)
AND
DB_AMP_Shard_RecoveryMap(_MarkerStatus, _Template)
AND
HasActiveStatus(_Item, _MarkerStatus, 1)
AND
NOT DB_AMP_Shard_Count(_Item, _MarkerStatus, _)
AND
NOT DB_AMP_Plus_Lamb_Gifts(_Item, _MarkerStatus, _)
THEN
TemplateAddTo(_Template, _Char, 1, 1);
RemoveStatus(_Item, _MarkerStatus, _Char);
PROC_AMP_RecoverTransferCleanup(_Char, _MarkerStatus);

// a lamb that has already paid out is stripped, not returned
PROC
PROC_AMP_Plus_RecoverFromSlot((CHARACTER)_Char, (STRING)_Slot)
AND
GetEquippedItem(_Char, _Slot, _Item)
AND
DB_AMP_Plus_Lamb_Gifts(_Item, _MarkerStatus, _)
AND

AND
NOT DB_AMP_Shard_Count(_Item, _MarkerStatus, _)
THEN
RemoveStatus(_Item, _MarkerStatus, _Char);
PROC_AMP_RecoverTransferCleanup(_Char, _MarkerStatus);

THEN

// spawn pass only reads the counter table
