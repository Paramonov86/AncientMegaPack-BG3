// Glory is the status's frozen duration. each rank has its own stack, so ranks never evict each other
// standing status grows only via its tick marker. Shown record stops the award chain's second pass adding

PROC
PROC_AMP_WarRank_ApplyRank((CHARACTER)_P, (INTEGER)_R)
AND

NOT DB_AMP_WarRank_Active(_P, _Old);

PROC
PROC_AMP_WarRank_ApplyRank((CHARACTER)_P, (INTEGER)_R)
THEN
PROC_AMP_WarRank_DrainShownOther(_P, _R);

PROC
PROC_AMP_WarRank_DrainShownOther((CHARACTER)_P, (INTEGER)_R)
AND
DB_AMP_WarRank_Shown(_P, (INTEGER)_Other, (INTEGER)_N)
AND
_Other != _R
AND
IntegerToString(_Other, (STRING)_OtherStr)
AND
Concatenate("AMP_WARRANK_R", _OtherStr, (STRING)_OtherStatus)
THEN
NOT DB_AMP_WarRank_Shown(_P, _Other, _N);
RemoveStatus(_P, _OtherStatus, NULL_00000000-0000-0000-0000-000000000000);
PROC_AMP_WarRank_DrainShownOther(_P, _R);

PROC
PROC_AMP_WarRank_ApplyRank((CHARACTER)_P, (INTEGER)_R)
AND
NOT DB_AMP_WarRank_Shown(_P, _R, _)
AND
IntegerToString(_R, (STRING)_RStr)
AND
Concatenate("AMP_WARRANK_R", _RStr, (STRING)_Status)
AND
HasActiveStatus(_P, _Status, 1)
AND
GetStatusTurns(_P, _Status, (INTEGER)_Turns)
THEN
DB_AMP_WarRank_Shown(_P, _R, _Turns);

PROC
PROC_AMP_WarRank_ApplyRank((CHARACTER)_P, (INTEGER)_R)
AND
_R >= 1
AND
NOT DB_AMP_WarRank_Shown(_P, _R, _)
AND
DB_AMP_WarRank_Glory(_P, (INTEGER)_Glory)
AND
_Glory > 0
AND
IntegerToString(_R, (STRING)_RStr)
AND
Concatenate("AMP_WARRANK_R", _RStr, (STRING)_Status)
AND
HasActiveStatus(_P, _Status, 0)
AND
IntegerProduct(_Glory, 6, (INTEGER)_Secs)
AND
IntegerToReal(_Secs, (REAL)_Dur)
THEN
DB_AMP_WarRank_Shown(_P, _R, _Glory);
ApplyStatus(_P, _Status, _Dur, 1, _P);

PROC
PROC_AMP_WarRank_ApplyRank((CHARACTER)_P, (INTEGER)_R)
AND
DB_AMP_WarRank_Shown(_P, _R, (INTEGER)_N)
AND
DB_AMP_WarRank_Glory(_P, (INTEGER)_Glory)
AND
_Glory > _N
AND
IntegerSum(_N, 1, (INTEGER)_Next)
AND
IntegerToString(_R, (STRING)_RStr)
AND
Concatenate("AMP_WARRANK_TICK_R", _RStr, (STRING)_Tick)
THEN
NOT DB_AMP_WarRank_Shown(_P, _R, _N);
DB_AMP_WarRank_Shown(_P, _R, _Next);
ApplyStatus(_P, _Tick, 0.0, 1, _P);

PROC
PROC_AMP_WarRank_ApplyRank((CHARACTER)_P, (INTEGER)_R)
AND
_R >= 1
THEN
DB_AMP_WarRank_Active(_P, _R);
