LevelGameplayStarted(_HibLvl, _HibEditor)

NOT DB_AMP_Feral_Disabled(1)

CombatStarted((GUIDSTRING)_C)
AND
DB_AMP_Difficulty_Mode("IMPOSSIBLE")
THEN

AND
DB_AMP_Difficulty_Mode("IMPOSSIBLE")

AND

AND
DB_AMP_Feral_Member(_E, (GUIDSTRING)_C)

// resolver cannot revert a bump, so off means strip and recompute
PROC
PROC_AMP_Plus_Feral_Regrade()
THEN
PROC_AMP_UpdateAvgPartyLevel();
PROC_AMP_UpdateCurrentTier();
PROC_AMP_DiffV3_RetierAll_Collect();
PROC_AMP_DiffV3_RetierAll_Strip();
PROC_AMP_Plus_Feral_ClearBumpMarkers();
PROC_AMP_DiffV3_RetierAll_Reapply();

PROC
PROC_AMP_Plus_Feral_ClearBumpMarkers()
THEN
PROC_AMP_Plus_Feral_DrainTierBump();

PROC
PROC_AMP_Plus_Feral_ClearBumpMarkers()
THEN
PROC_AMP_Plus_Feral_DrainAvariceBumped();

// single pass, the opener re-adds retracted markers
PROC
PROC_AMP_Plus_Feral_DrainTierBump()
AND
DB_AMP_Plus_TierBump((CHARACTER)_C, (STRING)_B, (STRING)_Ap)
THEN
NOT DB_AMP_Plus_TierBump(_C, _B, _Ap);

PROC
PROC_AMP_Plus_Feral_DrainAvariceBumped()
AND
DB_AMP_Plus_AvariceBumped((CHARACTER)_C, (STRING)_O, (STRING)_Bp)
THEN
NOT DB_AMP_Plus_AvariceBumped(_C, _O, _Bp);
PROC_AMP_Plus_Feral_DrainAvariceBumped();
