// drop stale spore-carrier links where the item is gone or no longer the spore template.
// act transitions / save-load / console deletes leave these behind -> phantom infections on rest.

// at rest start, before the main script reads links on finish
IF
LongRestStarted()
AND
DB_AMP_Spore_Carrier((CHARACTER)_Player, (ITEM)_SporeItem)
AND
NOT GetTemplate(_SporeItem, (ITEMROOT)AMP_Cursed_Spore_ec2ffd1d-2f0f-420a-989d-ea6d1c5bfee9)
THEN
NOT DB_AMP_Spore_Carrier(_Player, _SporeItem);

// also at level start so existing saves get cleaned without waiting for a rest
IF
LevelGameplayStarted(_, _)
AND
DB_AMP_Spore_Carrier((CHARACTER)_Player, (ITEM)_SporeItem)
AND
NOT GetTemplate(_SporeItem, (ITEMROOT)AMP_Cursed_Spore_ec2ffd1d-2f0f-420a-989d-ea6d1c5bfee9)
THEN
NOT DB_AMP_Spore_Carrier(_Player, _SporeItem);

EXITSECTION
ENDEXITSECTION
