IF
StatusApplied((CHARACTER)_Boss, "AMP_MIRROR_TRIGGER", _, _)
AND
NOT DB_AMP_MirrorSupport_Fired(_Boss)
AND
NOT DB_AMP_MirrorChase_Mob(_, _)
AND
IsBoss(_Boss, 1)
AND
IsInCombat(_Boss, 1)
AND
IsDead(_Boss, 0)
AND
GetHostCharacter((CHARACTER)_Host)
AND
IsEnemy(_Boss, _Host, 1)
AND
DB_AMP_MirrorPool_Size((INTEGER)_PoolSize)
AND
_PoolSize > 0
THEN
DB_AMP_MirrorSupport_Fired(_Boss);
PROC_AMP_MirrorSupport_Spawn(_Boss);

// spawn dispatcher: pick (party-size) distinct pool bosses, clamped to pool size + 8
// ring slots. distinct via random start + window. each pick reuses spawnsolo.

PROC
PROC_AMP_MirrorSupport_Spawn((CHARACTER)_Boss)
AND
DB_AMP_MirrorPool_Size((INTEGER)_Size)
AND
CombatGetGuidFor(_Boss, (GUIDSTRING)_CombatGuid)
AND
CombatGetInvolvedPartyMembersCount(_CombatGuid, (INTEGER)_PartyN)
AND
GetPosition(_Boss, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
Random(_Size, (INTEGER)_Start)
THEN
PROC_AMP_MirrorSupport_CountReal(_Boss, _CombatGuid, _PartyN, 1, 0, _Size, _Start, _BX, _BY, _BZ);

// count only real (non-summon) party members. player summons (familiars, conjures,
// acid slimes) must not inflate the spawn count. idx is 1-based over involved members.

// done iterating -> hand the real count to resolvecount
PROC
PROC_AMP_MirrorSupport_CountReal((CHARACTER)_Boss, (GUIDSTRING)_CombatGuid, (INTEGER)_Total, (INTEGER)_Idx, (INTEGER)_Acc, (INTEGER)_Size, (INTEGER)_Start, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_Idx > _Total
THEN
PROC_AMP_MirrorSupport_ResolveCount(_Boss, _Acc, _Size, _Start, _BX, _BY, _BZ);

// member is not a summon -> count it, advance
PROC
PROC_AMP_MirrorSupport_CountReal((CHARACTER)_Boss, (GUIDSTRING)_CombatGuid, (INTEGER)_Total, (INTEGER)_Idx, (INTEGER)_Acc, (INTEGER)_Size, (INTEGER)_Start, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_Idx <= _Total
AND
CombatGetInvolvedPartyMember(_CombatGuid, _Idx, (CHARACTER)_Member)
AND
IsSummon(_Member, 0)
AND
IntegerSum(_Acc, 1, (INTEGER)_Acc2)
AND
IntegerSum(_Idx, 1, (INTEGER)_Idx2)
THEN
PROC_AMP_MirrorSupport_CountReal(_Boss, _CombatGuid, _Total, _Idx2, _Acc2, _Size, _Start, _BX, _BY, _BZ);

// member is a summon -> skip, advance without counting
PROC
PROC_AMP_MirrorSupport_CountReal((CHARACTER)_Boss, (GUIDSTRING)_CombatGuid, (INTEGER)_Total, (INTEGER)_Idx, (INTEGER)_Acc, (INTEGER)_Size, (INTEGER)_Start, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_Idx <= _Total
AND
CombatGetInvolvedPartyMember(_CombatGuid, _Idx, (CHARACTER)_Member)
AND
IsSummon(_Member, 1)
AND
IntegerSum(_Idx, 1, (INTEGER)_Idx2)
THEN
PROC_AMP_MirrorSupport_CountReal(_Boss, _CombatGuid, _Total, _Idx2, _Acc, _Size, _Start, _BX, _BY, _BZ);

// clamp wanted count to pool size
PROC
PROC_AMP_MirrorSupport_ResolveCount((CHARACTER)_Boss, (INTEGER)_PartyN, (INTEGER)_Size, (INTEGER)_Start, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_PartyN <= _Size
THEN
PROC_AMP_MirrorSupport_CapEight(_Boss, _PartyN, _Size, _Start, _BX, _BY, _BZ);

PROC
PROC_AMP_MirrorSupport_ResolveCount((CHARACTER)_Boss, (INTEGER)_PartyN, (INTEGER)_Size, (INTEGER)_Start, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_PartyN > _Size
THEN
PROC_AMP_MirrorSupport_CapEight(_Boss, _Size, _Size, _Start, _BX, _BY, _BZ);

// clamp to 8 ring slots, seed loop state
PROC
PROC_AMP_MirrorSupport_CapEight((CHARACTER)_Boss, (INTEGER)_Count, (INTEGER)_Size, (INTEGER)_Start, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_Count <= 8
THEN
DB_AMP_MirrorSpawn_State(_Boss, _Start, _Size, _Count, 0, _BX, _BY, _BZ);
PROC_AMP_MirrorSupport_Loop(_Boss);

PROC
PROC_AMP_MirrorSupport_CapEight((CHARACTER)_Boss, (INTEGER)_Count, (INTEGER)_Size, (INTEGER)_Start, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_Count > 8
THEN
DB_AMP_MirrorSpawn_State(_Boss, _Start, _Size, 8, 0, _BX, _BY, _BZ);
PROC_AMP_MirrorSupport_Loop(_Boss);

// loop k=0..count-1: idx=(start+k) mod size -> distinct entry -> spawn at slot k
PROC
PROC_AMP_MirrorSupport_Loop((CHARACTER)_Boss)
AND
DB_AMP_MirrorSpawn_State(_Boss, (INTEGER)_Start, (INTEGER)_Size, (INTEGER)_Count, (INTEGER)_K, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_K < _Count
AND
IntegerSum(_Start, _K, (INTEGER)_Tmp)
AND
IntegerModulo(_Tmp, _Size, (INTEGER)_Idx)
AND
DB_AMP_MirrorPool(_Idx, (GUIDSTRING)_Template, (INTEGER)_PoolCount)
AND
NOT DB_AMP_MirrorBoss_Disabled(_Idx)
AND
IntegerSum(_K, 1, (INTEGER)_K2)
THEN
PROC_AMP_MirrorSupport_SpawnPick(_Boss, _Template, _PoolCount, _K, _BX, _BY, _BZ);
NOT DB_AMP_MirrorSpawn_State(_Boss, _Start, _Size, _Count, _K, _BX, _BY, _BZ);
DB_AMP_MirrorSpawn_State(_Boss, _Start, _Size, _Count, _K2, _BX, _BY, _BZ);
PROC_AMP_MirrorSupport_Loop(_Boss);

// disabled pool entry (toggled off in settings) -> advance k, no spawn
PROC
PROC_AMP_MirrorSupport_Loop((CHARACTER)_Boss)
AND
DB_AMP_MirrorSpawn_State(_Boss, (INTEGER)_Start, (INTEGER)_Size, (INTEGER)_Count, (INTEGER)_K, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_K < _Count
AND
IntegerSum(_Start, _K, (INTEGER)_TmpD)
AND
IntegerModulo(_TmpD, _Size, (INTEGER)_IdxD)
AND
DB_AMP_MirrorBoss_Disabled(_IdxD)
AND
IntegerSum(_K, 1, (INTEGER)_K2D)
THEN
NOT DB_AMP_MirrorSpawn_State(_Boss, _Start, _Size, _Count, _K, _BX, _BY, _BZ);
DB_AMP_MirrorSpawn_State(_Boss, _Start, _Size, _Count, _K2D, _BX, _BY, _BZ);
PROC_AMP_MirrorSupport_Loop(_Boss);

PROC
PROC_AMP_MirrorSupport_Loop((CHARACTER)_Boss)
AND
DB_AMP_MirrorSpawn_State(_Boss, (INTEGER)_Start, (INTEGER)_Size, (INTEGER)_Count, (INTEGER)_K, (REAL)_BX, (REAL)_BY, (REAL)_BZ)
AND
_K >= _Count
THEN
NOT DB_AMP_MirrorSpawn_State(_Boss, _Start, _Size, _Count, _K, _BX, _BY, _BZ);
