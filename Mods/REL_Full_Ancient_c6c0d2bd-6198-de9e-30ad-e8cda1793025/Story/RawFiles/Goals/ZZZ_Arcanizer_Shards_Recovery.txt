// RECOVERY SHARD - Use trigger
IF
TemplateUseStarted((CHARACTER)_Char, (ITEMROOT)_Template, (ITEM)_Instance)
AND
DB_AMP_Shard_Recovery(_Template)
THEN
PROC_AMP_RecoverAllShards(_Char);
PlaySound(_Char, "CrSpell_Cast_MagicClub");
RequestDelete(_Instance);

// Iterate all equipment slots
PROC
PROC_AMP_RecoverAllShards((CHARACTER)_Char)
AND
DB_AMP_Shard_RecoverySlot(_Slot)
THEN
PROC_AMP_RecoverFromSlot(_Char, _Slot);

// NonStack recovery: check each marker on item
PROC
PROC_AMP_RecoverFromSlot((CHARACTER)_Char, (STRING)_Slot)
AND
GetEquippedItem(_Char, _Slot, _Item)
AND
DB_AMP_Shard_RecoveryMap(_MarkerStatus, _Template)
AND
HasActiveStatus(_Item, _MarkerStatus, 1)
AND
NOT DB_AMP_Shard_Count(_Item, _MarkerStatus, _)
THEN
TemplateAddTo(_Template, _Char, 1, 1);
RemoveStatus(_Item, _MarkerStatus, _Char);
PROC_AMP_RecoverTransferCleanup(_Char, _MarkerStatus);

// stackable recovery: drain the shard-count table one row per pass.
// retracting while iterating truncated after the first row, so 2nd/3rd shards got left behind.
// now: handle one row, retract, recurse until empty.
PROC
PROC_AMP_RecoverFromSlot((CHARACTER)_Char, (STRING)_Slot)
AND
GetEquippedItem(_Char, _Slot, _Item)
THEN
PROC_AMP_RecoverStackableDrain(_Char, _Item);

PROC
PROC_AMP_RecoverStackableDrain((CHARACTER)_Char, (ITEM)_Item)
AND
DB_AMP_Shard_Count(_Item, _MarkerStatus, _Count)
AND
DB_AMP_Shard_RecoveryMap(_MarkerStatus, _Template)
THEN
NOT DB_AMP_Shard_Count(_Item, _MarkerStatus, _Count);
PROC_AMP_SpawnMultiple(_Char, _Template, _Count);
PROC_AMP_RecoverTransferCleanup(_Char, _MarkerStatus);
RemoveStatus(_Item, _MarkerStatus, _Char);
PROC_AMP_Shard_DecTotalExtra(_Item, _Count);
PROC_AMP_RecoverStackableDrain(_Char, _Item);

// Transfer cleanup: remove CHAR status
PROC
PROC_AMP_RecoverTransferCleanup((CHARACTER)_Char, (STRING)_MarkerStatus)
AND
DB_AMP_Shard_Transfer(_MarkerStatus, _CharStatus)
THEN
RemoveStatus(_Char, _CharStatus, _Char);

// Transfer cleanup: remove external vanilla statuses
PROC
PROC_AMP_RecoverTransferCleanup((CHARACTER)_Char, (STRING)_MarkerStatus)
AND
DB_AMP_Shard_Transfer(_MarkerStatus, _CharStatus)
AND
DB_AMP_Shard_ExternalStatus(_CharStatus, _ExternalStatus)
THEN
RemoveStatus(_Char, _ExternalStatus, _Char);

// Spawn N copies helper
PROC
PROC_AMP_SpawnMultiple((CHARACTER)_Char, (ITEMROOT)_Template, (INTEGER)_Count)
AND
_Count > 0
AND
IntegerSubtract(_Count, 1, (INTEGER)_Next)
THEN
TemplateAddTo(_Template, _Char, 1, 1);
PROC_AMP_SpawnMultiple(_Char, _Template, _Next);

PROC
PROC_AMP_SpawnMultiple((CHARACTER)_Char, (ITEMROOT)_Template, (INTEGER)_Count)
AND
_Count <= 0
THEN
DB_NOOP(1);

// Wipe all Shard_Count rows for (Item, Status)
PROC
PROC_AMP_Shard_ClearCount((ITEM)_Item, (STRING)_Status)
AND
DB_AMP_Shard_Count(_Item, _Status, _Any)
THEN
NOT DB_AMP_Shard_Count(_Item, _Status, _Any);

// Decrement TotalCount (_N - 1) extra times
PROC
PROC_AMP_Shard_DecTotalExtra((ITEM)_Item, (INTEGER)_N)
AND
_N > 1
AND
IntegerSubtract(_N, 1, (INTEGER)_Next)
THEN
PROC_AMP_Shard_DecTotal(_Item);
PROC_AMP_Shard_DecTotalExtra(_Item, _Next);

PROC
PROC_AMP_Shard_DecTotalExtra((ITEM)_Item, (INTEGER)_N)
AND
_N <= 1
THEN
DB_NOOP(1);
