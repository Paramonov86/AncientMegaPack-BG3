// apply status n times (stackable shards)
PROC
PROC_AMP_ApplyShardEffectN((CHARACTER)_Char, (STRING)_Status, (INTEGER)_Count)
AND
_Count > 0
AND
IntegerSubtract(_Count, 1, _NewCount)
THEN
ApplyStatus(_Char, _Status, -1.0, 1, _Char);
PROC_AMP_ApplyShardEffectN(_Char, _Status, _NewCount);

PROC
PROC_AMP_ApplyShardEffectN((CHARACTER)_Char, (STRING)_Status, (INTEGER)_Count)
AND
_Count <= 0
THEN
DB_NOOP(1);

// weapon shards, stackable

// melee equipped - ok
IF
TemplateUseStarted((CHARACTER)_Char, (ITEMROOT)_Template, (ITEM)_Instance)
AND
HasActiveStatus(_Instance, "TOT_SHOP_ITEM", 0)
AND
DB_AMP_Shard_Weapon_Stackable(_Template, _MeleeStatus, _)
AND
HasMeleeWeaponEquipped(_Char, "Mainhand", 1)
AND
GetEquippedItem(_Char, "Melee Main Weapon", _Weapon)
AND
QRY_AMP_Shard_CanApply(_Weapon)
THEN
PROC_AMP_Shard_IncTotal(_Weapon);
ApplyStatus(_Char, _MeleeStatus, 0.0, 1, _Char);
PlaySound(_Char, "CrSpell_Cast_MagicClub");
RequestDelete(_Instance);

// ranged fallback - ok
IF
TemplateUseStarted((CHARACTER)_Char, (ITEMROOT)_Template, (ITEM)_Instance)
AND
HasActiveStatus(_Instance, "TOT_SHOP_ITEM", 0)
AND
DB_AMP_Shard_Weapon_Stackable(_Template, _, _RangedStatus)
AND
HasMeleeWeaponEquipped(_Char, "Mainhand", 0)
AND
HasRangedWeaponEquipped(_Char, "Mainhand", 1)
AND
GetEquippedItem(_Char, "Ranged Main Weapon", _Weapon)
AND
QRY_AMP_Shard_CanApply(_Weapon)
THEN

// status transfer system

// statusapplied - increment count and apply effect
IF
StatusApplied((ITEM)_Item, _ItemStatus, _Causee, _)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _CharStatus)
AND
IsEquipped(_Item, 1)
AND
GetInventoryOwner(_Item, _Char)
AND
DB_AMP_Shard_Count(_Item, _ItemStatus, _OldCount)
AND
IntegerSum(_OldCount, 1, _NewCount)
THEN
NOT DB_AMP_Shard_Count(_Item, _ItemStatus, _OldCount);
DB_AMP_Shard_Count(_Item, _ItemStatus, _NewCount);
ApplyStatus(_Char, _CharStatus, -1.0, 1, _Char);

// statusapplied - first application
IF
StatusApplied((ITEM)_Item, _ItemStatus, _Causee, _)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _CharStatus)
AND
IsEquipped(_Item, 1)
AND
GetInventoryOwner(_Item, _Char)
AND
NOT DB_AMP_Shard_Count(_Item, _ItemStatus, _)
THEN
DB_AMP_Shard_Count(_Item, _ItemStatus, 1);
ApplyStatus(_Char, _CharStatus, -1.0, 1, _Char);

// equipped - apply n times based on count
// item-keyed db_amp_shard_count is matched first so a non-shard equip exits
// before joining the full db_amp_shard_transfer table (avoids per-equip cross-join).
IF
Equipped((ITEM)_Item, (CHARACTER)_Char)
AND
DB_AMP_Shard_Count(_Item, _ItemStatus, _Count)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _CharStatus)
AND
HasActiveStatus(_Item, _ItemStatus, 1)
THEN
PROC_AMP_ApplyShardEffectN(_Char, _CharStatus, _Count);

// unequipped - remove all effects
IF
Unequipped((ITEM)_Item, (CHARACTER)_Char)
AND
DB_AMP_Shard_Count(_Item, _ItemStatus, _)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _CharStatus)
THEN
RemoveStatus(_Char, _CharStatus, _Char);

// unequipped - remove external vanilla statuses
IF
Unequipped((ITEM)_Item, (CHARACTER)_Char)
AND
DB_AMP_Shard_Count(_Item, _ItemStatus, _)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _CharStatus)
AND
DB_AMP_Shard_ExternalStatus(_CharStatus, _ExternalStatus)
THEN
RemoveStatus(_Char, _ExternalStatus, _Char);

// statusremoved - decrement count
IF
StatusRemoved((ITEM)_Item, _ItemStatus, _, _)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _)
AND
DB_AMP_Shard_Count(_Item, _ItemStatus, _Count)
AND
_Count > 1
AND
IntegerSubtract(_Count, 1, _NewCount)
THEN
NOT DB_AMP_Shard_Count(_Item, _ItemStatus, _Count);
DB_AMP_Shard_Count(_Item, _ItemStatus, _NewCount);
PROC_AMP_Shard_DecTotal(_Item);

// statusremoved - last one, clear count
IF
StatusRemoved((ITEM)_Item, _ItemStatus, _, _)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _)
AND
DB_AMP_Shard_Count(_Item, _ItemStatus, 1)
THEN
NOT DB_AMP_Shard_Count(_Item, _ItemStatus, 1);
PROC_AMP_Shard_DecTotal(_Item);
