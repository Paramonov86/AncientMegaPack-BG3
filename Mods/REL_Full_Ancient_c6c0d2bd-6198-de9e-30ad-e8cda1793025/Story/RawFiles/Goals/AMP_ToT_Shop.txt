// enough gold, deduct once
PROC

AND
_Total > 0
AND
NOT DB_ToT_Shop_DeductDone(1)
AND
DB_ToT_Shop_PendingPurchase((ITEM)_AnyItem, _, _, _)
AND

RemoveGoldFromMagicPockets(_Player, _Total);
PlaySound(_Player, "UI_Gameplay_Coin_Removed");

PROC_ToT_Shop_DeductBatchGold((INTEGER)_Total)
AND
_Total > 0
AND
NOT DB_ToT_Shop_DeductDone(1)
AND
DB_ToT_Shop_PendingPurchase((ITEM)_AnyItem, _, _, _)
AND
GetInventoryOwner(_AnyItem, (CHARACTER)_Player)
AND
DB_Players(_Player)
AND
GetGoldInMagicPockets(_Player, (INTEGER)_Gold)
AND
_Gold < _Total
THEN
DB_ToT_Shop_DeductDone(1);
DB_ToT_Shop_BatchReturn(1);
PlaySound(_Player, "VFX_Status_WildMagic_Barbarian_Flumph_SpawnImpact");
PROC_ToT_Shop_ShowNoGoldMessage(_Player);

// one weighted pick with dedup re-roll, capped at 20 attempts

PROC
PROC_ToT_Shop_PickWeighted((STRING)_ShopID, (ITEM)_Container, (INTEGER)_Attempt)
AND
_Attempt < 20
AND
DB_ToT_Shop_TotalWeight(_ShopID, (INTEGER)_TotalW)
AND
_TotalW > 0
AND
Random(_TotalW, (INTEGER)_Roll)
THEN
PROC_ToT_PickCategory(_ShopID, _Roll);
PROC_ToT_CheckDedup(_ShopID, _Container, _Attempt);

// hit the cap, skip this slot
PROC
PROC_ToT_Shop_PickWeighted((STRING)_ShopID, (ITEM)_Container, (INTEGER)_Attempt)
AND
_Attempt >= 20
THEN
PROC_ToT_Shop_AdvanceSpawnCounter(_ShopID, _Container);

// unique spawns, duplicate re-rolls

// unique, spawn and advance
PROC
PROC_ToT_CheckDedup((STRING)_ShopID, (ITEM)_Container, (INTEGER)_Attempt)
AND
DB_ToT_PickResult(_ShopID, (ITEMROOT)_Template)
AND
NOT DB_ToT_Shop_SelectedThisRestock(_Container, _Template)
THEN
NOT DB_ToT_PickResult(_ShopID, _Template);
DB_ToT_Shop_SelectedThisRestock(_Container, _Template);
PROC_ToT_Shop_SpawnItem(_ShopID, _Container, _Template);
PROC_ToT_Shop_AdvanceSpawnCounter(_ShopID, _Container);

// duplicate, re-roll
PROC
PROC_ToT_CheckDedup((STRING)_ShopID, (ITEM)_Container, (INTEGER)_Attempt)
AND
DB_ToT_PickResult(_ShopID, (ITEMROOT)_Template)
AND
DB_ToT_Shop_SelectedThisRestock(_Container, _Template)
AND
IntegerSum(_Attempt, 1, (INTEGER)_Next)
THEN
NOT DB_ToT_PickResult(_ShopID, _Template);
PROC_ToT_Shop_PickWeighted(_ShopID, _Container, _Next);

// spawn: add to container, mark pending, sound
PROC
PROC_ToT_Shop_SpawnItem((STRING)_ShopID, (ITEM)_Container, (ITEMROOT)_Template)
AND
GetHostCharacter((CHARACTER)_Host)
THEN
DB_ToT_Shop_PendingSpawn(_Container, _Template);
TemplateAddTo(_Template, _Container, 1, 0);
PlaySound(_Host, "Spell_Impact_Wizard_BenignTranspositionTeleport_L1to3");

// count++ then check done
PROC
PROC_ToT_Shop_AdvanceSpawnCounter((STRING)_ShopID, (ITEM)_Container)
AND
DB_ToT_Shop_SpawnCounter(_ShopID, (INTEGER)_Count)
AND
IntegerSum(_Count, 1, (INTEGER)_NewCount)
THEN
NOT DB_ToT_Shop_SpawnCounter(_ShopID, _Count);
DB_ToT_Shop_SpawnCounter(_ShopID, _NewCount);
PROC_ToT_Shop_CheckRestockDone(_ShopID, _Container, _NewCount);

// more needed -> next tick, else cleanup

// need more, next tick
PROC
PROC_ToT_Shop_CheckRestockDone((STRING)_ShopID, (ITEM)_Container, (INTEGER)_Count)
AND
DB_ToT_Shop_MaxStock(_ShopID, (INTEGER)_Max)
AND
_Count < _Max
AND
GetHostCharacter((CHARACTER)_Host)
THEN
ObjectTimerLaunch(_Host, "ToT_Shop_SpawnNext", 200, 0);

// done, cleanup
PROC
PROC_ToT_Shop_CheckRestockDone((STRING)_ShopID, (ITEM)_Container, (INTEGER)_Count)
AND
DB_ToT_Shop_MaxStock(_ShopID, (INTEGER)_Max)
AND
_Count >= _Max
THEN
PROC_ToT_Shop_CleanupRestockTemp(_ShopID, _Container);
