// core inserts DB_ToT_Shop_Activated per container without clearing the opposite,
// so after a few toggles both (1) and (0) sit in the db and the shop never really
// turns off. sysclear first, then write the single correct value.

IF
MessageBoxYesNoClosed((CHARACTER)_Player, "AMP_Settings018", 1)
THEN
SysClear("DB_ToT_Shop_Activated", 1);
DB_ToT_Shop_Activated(1);

IF
MessageBoxYesNoClosed((CHARACTER)_Player, "AMP_Settings018", 0)
THEN
SysClear("DB_ToT_Shop_Activated", 1);
DB_ToT_Shop_Activated(0);

EXITSECTION
ENDEXITSECTION
