// shard transfer fix v3.
// v2 missed amulet and ring2 (and its rules are frozen on old saves). v3 does the
// full clean + reapply across all slots including amulet/ring2.
// 7s timer runs after v2's 5s and overwrites its incomplete reapply.

IF
LevelGameplayStarted(_Level, _)
THEN
TimerLaunch("AMP_ShardFix3", 7000);

IF
TimerFinished("AMP_ShardFix3")
AND
DB_Players(_Player)
THEN
PROC_AMP_ShardFix3_Clean(_Player);
PROC_AMP_ShardFix3_Slot(_Player, "Breast");
PROC_AMP_ShardFix3_Slot(_Player, "Ring");
PROC_AMP_ShardFix3_Slot(_Player, "Ring2");
PROC_AMP_ShardFix3_Slot(_Player, "Gloves");
PROC_AMP_ShardFix3_Slot(_Player, "Helmet");
PROC_AMP_ShardFix3_Slot(_Player, "Cloak");
PROC_AMP_ShardFix3_Slot(_Player, "Boots");
PROC_AMP_ShardFix3_Slot(_Player, "Amulet");
PROC_AMP_ShardFix3_Slot(_Player, "Melee Main Weapon");
PROC_AMP_ShardFix3_Slot(_Player, "Ranged Main Weapon");

// phase 1: strip all transfer CHAR statuses off the character
PROC
PROC_AMP_ShardFix3_Clean((CHARACTER)_Char)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _CharStatus)
AND
HasActiveStatus((CHARACTER)_Char, _CharStatus, 1)
THEN
RemoveStatus(_Char, _CharStatus, _Char);

// phase 2: reapply per equipped slot
PROC
PROC_AMP_ShardFix3_Slot((CHARACTER)_Char, (STRING)_Slot)
AND
GetEquippedItem(_Char, _Slot, (ITEM)_Item)
AND
DB_AMP_Shard_Count(_Item, _ItemStatus, _Count)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _CharStatus)
AND
HasActiveStatus(_Item, _ItemStatus, 1)
THEN
PROC_AMP_ApplyShardEffectN(_Char, _CharStatus, _Count);

EXITSECTION
ENDEXITSECTION
