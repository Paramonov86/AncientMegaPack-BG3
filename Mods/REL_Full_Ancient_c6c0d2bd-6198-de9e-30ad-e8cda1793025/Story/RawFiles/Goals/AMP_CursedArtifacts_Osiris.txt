AND
DB_Players((CHARACTER)_Char)
AND
NOT DB_AMP_Elixir_Level(_Char, _)
THEN

IF
StatusApplied(_Char, "AMP_ELIXIR_UPGRADE_TRIGGER", _, _)
AND
DB_Players((CHARACTER)_Char)
AND
DB_AMP_Elixir_Level(_Char, (INTEGER)_Level)
AND
_Level < 10
AND
IntegerSum(_Level, 1, _NewLevel)
THEN
NOT DB_AMP_Elixir_Level(_Char, _Level);

PROC_AMP_Elixir_SetLevel(_Char, _NewLevel);

// run passes for one source, lifetime cap 4
PROC
PROC_AMP_Spore_RunSourceMulti((CHARACTER)_Player, (ITEM)_Source, (INTEGER)_Remaining)
AND
_Remaining > 0
AND
DB_AMP_Spore_SourceCap(_Source, (INTEGER)_Used)
AND
_Used < 4
AND
IntegerSubtract(_Remaining, 1, _NextRemaining)
THEN
PROC_AMP_Spore_RunSourcePass(_Player, _Source);
PROC_AMP_Spore_RunSourceMulti(_Player, _Source, _NextRemaining);

// same, source not tracked yet
PROC
PROC_AMP_Spore_RunSourceMulti((CHARACTER)_Player, (ITEM)_Source, (INTEGER)_Remaining)
AND
_Remaining > 0
AND
NOT DB_AMP_Spore_SourceCap(_Source, _)
AND
IntegerSubtract(_Remaining, 1, _NextRemaining)
THEN
DB_AMP_Spore_SourceCap(_Source, 0);
PROC_AMP_Spore_RunSourcePass(_Player, _Source);
PROC_AMP_Spore_RunSourceMulti(_Player, _Source, _NextRemaining);

// single pass: equipped slots first, then inventory
PROC
PROC_AMP_Spore_RunSourcePass((CHARACTER)_Player, (ITEM)_Source)
THEN
DB_AMP_Spore_CurrentSource(_Player, _Source);
// phase 1: equipped slots
PROC_AMP_Spore_InfectEquippedSlots(_Player, _Source);
// phase 2: inventory, skip equipped
IterateInventory(_Player, "AMP_Spore_TryInfect_Inv", "AMP_Spore_PassDone");

// phase 1: equipped slots, 0.5% each, cap 3 per rest, lifetime cap 4
PROC
PROC_AMP_Spore_InfectEquippedSlots((CHARACTER)_Player, (ITEM)_Source)
AND
NOT HasPassive(_Player, "MAG_Cloak30_bonus", 1)
AND
NOT HasActiveStatus(_Player, "AMP_GRAND_DESIGN_WARD", 1)
AND
DB_AMP_Spore_EquipSlots((STRING)_Slot)
AND
GetEquippedItem(_Player, _Slot, (ITEM)_Item)
AND
_Item != _Source
AND
DB_AMP_Spore_PassNewInfections(_Player, (INTEGER)_PassCount)
AND
_PassCount < 3
AND
DB_AMP_Spore_SourceCap(_Source, (INTEGER)_SourceUsed)
AND
_SourceUsed < 4
AND
NOT GetTemplate(_Item, (ITEMROOT)AMP_Cursed_Spore_ec2ffd1d-2f0f-420a-989d-ea6d1c5bfee9)
AND
NOT IsStoryItem(_Item, 1)
AND
Random(1000, (INTEGER)_Roll)
AND
_Roll < 5
AND
IntegerSum(_PassCount, 1, _NewPassCount)
AND
IntegerSum(_SourceUsed, 1, _NewSourceUsed)
THEN
ApplyStatus(_Item, "AMP_SPORE_INFECTION", -1.0, 1, _Player);
PROC_AMP_Spore_TrackInfection(_Item);
NOT DB_AMP_Spore_PassNewInfections(_Player, _PassCount);
DB_AMP_Spore_PassNewInfections(_Player, _NewPassCount);
NOT DB_AMP_Spore_SourceCap(_Source, _SourceUsed);
DB_AMP_Spore_SourceCap(_Source, _NewSourceUsed);
DB_AMP_Spore_EquippedChecked(_Item);

// mark equipped items so phase 2 skips them
PROC
PROC_AMP_Spore_InfectEquippedSlots((CHARACTER)_Player, (ITEM)_Source)
AND
DB_AMP_Spore_EquipSlots((STRING)_Slot)
AND
GetEquippedItem(_Player, _Slot, (ITEM)_Item)
AND
NOT DB_AMP_Spore_EquippedChecked(_Item)
THEN
DB_AMP_Spore_EquippedChecked(_Item);

// phase 2: inventory, 1%, skip equipped, same caps as phase 1
IF
EntityEvent((ITEM)_Item, "AMP_Spore_TryInfect_Inv")
AND
DB_AMP_Spore_CurrentSource((CHARACTER)_Player, (ITEM)_Source)
AND
NOT HasPassive(_Player, "MAG_Cloak30_bonus", 1)
AND
NOT HasActiveStatus(_Player, "AMP_GRAND_DESIGN_WARD", 1)
AND
_Item != _Source
AND
NOT DB_AMP_Spore_EquippedChecked(_Item)
AND
DB_AMP_Spore_PassNewInfections(_Player, (INTEGER)_PassCount)
AND
_PassCount < 3
AND
DB_AMP_Spore_SourceCap(_Source, (INTEGER)_SourceUsed)
AND
_SourceUsed < 4
AND
NOT GetTemplate(_Item, (ITEMROOT)AMP_Cursed_Spore_ec2ffd1d-2f0f-420a-989d-ea6d1c5bfee9)
AND
NOT IsStoryItem(_Item, 1)
AND
Random(1000, (INTEGER)_Roll)
AND
_Roll < 10
AND
IntegerSum(_PassCount, 1, _NewPassCount)
AND
IntegerSum(_SourceUsed, 1, _NewSourceUsed)
THEN
ApplyStatus(_Item, "AMP_SPORE_INFECTION", -1.0, 1, _Player);
PROC_AMP_Spore_TrackInfection(_Item);
NOT DB_AMP_Spore_PassNewInfections(_Player, _PassCount);
DB_AMP_Spore_PassNewInfections(_Player, _NewPassCount);
NOT DB_AMP_Spore_SourceCap(_Source, _SourceUsed);
DB_AMP_Spore_SourceCap(_Source, _NewSourceUsed);

// cleanup after pass
IF
EntityEvent(_, "AMP_Spore_PassDone")
AND
DB_AMP_Spore_CurrentSource((CHARACTER)_Player, (ITEM)_Source)
THEN
NOT DB_AMP_Spore_CurrentSource(_Player, _Source);

// scatter

// collect equipped items
PROC
PROC_AMP_Sovereign_ScatterEquipment((CHARACTER)_Char)
AND
GetPosition(_Char, (REAL)_X, (REAL)_Y, (REAL)_Z)
AND
DB_AMP_Sovereign_ScatterSlots((STRING)_Slot)
AND
GetEquippedItem(_Char, _Slot, (ITEM)_Item)
THEN
DB_AMP_Sovereign_ScatterQueue(_Char, _Item, _X, _Y, _Z);
RealtimeObjectTimerLaunch(_Char, "AMP_Sovereign_Scatter_Timer", 500);

// timer done: unequip
IF
ObjectTimerFinished(_Char, "AMP_Sovereign_Scatter_Timer")
AND
DB_AMP_Sovereign_ScatterQueue((CHARACTER)_Char, (ITEM)_Item, (REAL)_X, (REAL)_Y, (REAL)_Z)
THEN

// unequipped: scatter + item timer
IF
Unequipped(_Item, _Char)
AND
DB_AMP_Sovereign_ScatterQueue(_Char, _Item, (REAL)_X, (REAL)_Y, (REAL)_Z)
THEN
NOT DB_AMP_Sovereign_ScatterQueue(_Char, _Item, _X, _Y, _Z);
ScatterAt(_Item, _X, _Y, _Z, 10.0);
ApplyStatus(_Item, "AMP_SOVEREIGN_SCATTER_ITEM_TIMER", 30.0, 1, _Char);
DB_AMP_Sovereign_ScatteredItem(_Item, _Char);

// item timer expired: delete if not picked up
IF
StatusRemoved(_Item, "AMP_SOVEREIGN_SCATTER_ITEM_TIMER", _, _)
AND
DB_AMP_Sovereign_ScatteredItem((ITEM)_Item, (CHARACTER)_Char)
AND
NOT IsInInventoryOf(_Item, _Char, _)
THEN
RequestDelete(_Item);
NOT DB_AMP_Sovereign_ScatteredItem(_Item, _Char);

PROC_AMP_Spore_DebugInfectEquip(_Host);

PROC_AMP_Spore_DebugInfectEquip((CHARACTER)_Host)
AND
DB_AMP_Spore_EquipSlots((STRING)_Slot)
AND
GetEquippedItem(_Host, _Slot, (ITEM)_Item)
THEN
ApplyStatus(_Item, "AMP_SPORE_INFECTION", -1.0, 1, _Host);
PROC_AMP_Spore_TrackInfection(_Item);

// debug: purge spore infection from inventory

TextEvent("amp_spore_purge")
AND
GetHostCharacter((CHARACTER)_Host)
THEN
IterateInventory(_Host, "AMP_Spore_DebugPurge", "AMP_Spore_DebugPurgeDone");

IF
EntityEvent((ITEM)_Item, "AMP_Spore_DebugPurge")
AND
GetInventoryOwner(_Item, (CHARACTER)_Owner)
AND
HasActiveStatus(_Item, "AMP_SPORE_INFECTION", 1)
THEN
RemoveStatus(_Item, "AMP_SPORE_INFECTION", _Owner);

// force-clean db even if status was already engine-wiped
IF
EntityEvent((ITEM)_Item, "AMP_Spore_DebugPurge")
AND
DB_AMP_Spore_ItemInfections(_Item, (INTEGER)_Count)
THEN
NOT DB_AMP_Spore_ItemInfections(_Item, _Count);

IF
EntityEvent(_, "AMP_Spore_DebugPurgeDone")
THEN
DB_NOOP(1);

// debug: nuke all spores (party + camp chests + dbs)
IF
TextEvent("amp_spore_nuke")
THEN
PROC_AMP_Spore_NukeAll();

// delete tracked spore items, clear carrier link
PROC
PROC_AMP_Spore_NukeAll()
AND
DB_AMP_Spore_Carrier((CHARACTER)_Player, (ITEM)_SporeItem)
THEN
RequestDelete(_SporeItem);
NOT DB_AMP_Spore_Carrier(_Player, _SporeItem);

// strip infection status + db from tracked infected items
PROC
PROC_AMP_Spore_NukeAll()
AND
DB_AMP_Spore_ItemInfections((ITEM)_Item, (INTEGER)_Count)
THEN
RemoveStatus(_Item, "AMP_SPORE_INFECTION");
NOT DB_AMP_Spore_ItemInfections(_Item, _Count);

// sweep party inventories for untracked spores / stale infections
PROC
PROC_AMP_Spore_NukeAll()
AND
DB_Players((CHARACTER)_Player)
THEN
IterateInventory(_Player, "AMP_Spore_Nuke_Inv", "AMP_Spore_Nuke_Done");

DB_Camp_UserCampChest(_, (ITEM)_Chest)

DB_AMP_Spore_CurrentSource((CHARACTER)_P, (ITEM)_S)

NOT DB_AMP_Spore_CurrentSource(_P, _S);

DB_AMP_Spore_EquippedChecked((ITEM)_I)

NOT DB_AMP_Spore_EquippedChecked(_I);

DB_AMP_SporeChest_PassNew((ITEM)_C, (INTEGER)_N)

AND
DB_AMP_SporeChest_Current((ITEM)_C)
THEN
NOT DB_AMP_SporeChest_Current(_C);

// per-item nuke: kill spore items + leftover infection
IF
EntityEvent((ITEM)_Item, "AMP_Spore_Nuke_Inv")
AND
GetTemplate(_Item, (ITEMROOT)AMP_Cursed_Spore_ec2ffd1d-2f0f-420a-989d-ea6d1c5bfee9)

EntityEvent((ITEM)_Item, "AMP_Spore_Nuke_Inv")

HasActiveStatus(_Item, "AMP_SPORE_INFECTION", 1)
THEN

AND
DB_AMP_Spore_ItemInfections(_Item, (INTEGER)_Count)
THEN
NOT DB_AMP_Spore_ItemInfections(_Item, _Count);

EntityEvent(_, "AMP_Spore_Nuke_Done")
THEN
