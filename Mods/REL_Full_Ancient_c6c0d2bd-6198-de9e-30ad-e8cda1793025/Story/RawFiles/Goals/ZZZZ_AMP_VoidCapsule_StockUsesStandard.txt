// stock gamemode maps to difficulty NONE, same key as casual, and the capsule
// cache keys off that. stock should play like standard but casual keeps NONE,
// so override the active cache to standard values when gamemode is stock.
// 500ms delay so we run after refreshcache fires on the same event.

IF
LevelGameplayStarted(_, _)
AND
DB_AMP_GameMode("STOCK")
AND
GetHostCharacter((CHARACTER)_Host)
THEN
ObjectTimerLaunch(_Host, "AMP_Stock_CapsuleOverride", 500, 0);

IF
ObjectTimerFinished((CHARACTER)_Host, "AMP_Stock_CapsuleOverride")
AND
DB_AMP_GameMode("STOCK")
THEN
PROC_AMP_VoidCapsule_StockOverride();

// shard->coin chance: none 100 -> standard 25
PROC
PROC_AMP_VoidCapsule_StockOverride()
AND
DB_AMP_VoidCapsule_ShardCoinChanceActive((INTEGER)_Old)
AND
DB_AMP_VoidCapsule_ShardCoinChanceByMode("STANDARD", (INTEGER)_Std)
AND
_Old != _Std
THEN
NOT DB_AMP_VoidCapsule_ShardCoinChanceActive(_Old);
DB_AMP_VoidCapsule_ShardCoinChanceActive(_Std);
