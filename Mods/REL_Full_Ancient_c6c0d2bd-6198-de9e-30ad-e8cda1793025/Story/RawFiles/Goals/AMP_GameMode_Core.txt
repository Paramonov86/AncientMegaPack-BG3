// new game: auto-prompt on TUT_Avernus_C

IF
LevelGameplayStarted("TUT_Avernus_C", _)
AND
NOT DB_AMP_GameMode(_)
AND
NOT DB_AMP_Wizard_Exists(1)
AND
GetHostCharacter((CHARACTER)_Host)
THEN
TimerLaunch("AMP_GameMode_NewGame_Delay", 15000);

// existing save loaded on avernus - same prompt
IF
LevelGameplayStarted("TUT_Avernus_C", _)
AND
DB_AMP_Difficulty_Mode((STRING)_Mode)
AND
_Mode != "NONE"
AND
NOT DB_AMP_GameMode(_)
AND
NOT DB_AMP_Wizard_Exists(1)
AND
GetHostCharacter((CHARACTER)_Host)
THEN
TimerLaunch("AMP_GameMode_NewGame_Delay", 15000);

// training ring trigger

// not stock - normal flow
IF
UsingSpell((CHARACTER)_Caster, "Shout_AMP_Settings_GameMode", _, _, _)
AND
IsPlayer(_Caster, 1)
AND
GetHostCharacter(_Caster)
AND
NOT DB_AMP_GameMode("STOCK")
THEN
PROC_AMP_GameMode_ShowMB1(_Caster);

// stock - locked, just show the info box
IF
UsingSpell((CHARACTER)_Caster, "Shout_AMP_Settings_GameMode", _, _, _)
AND
IsPlayer(_Caster, 1)
AND
GetHostCharacter(_Caster)
AND
DB_AMP_GameMode("STOCK")
THEN
OpenMessageBox(_Caster, "AMP_Settings027");

// messagebox chain - normal/casual/hard/stock

// show mb1
PROC
PROC_AMP_GameMode_ShowMB1((CHARACTER)_Host)
THEN
OpenMessageBoxYesNo(_Host, "AMP_Settings023");

// mb1 normal: yes -> normal
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Settings023", 1)
THEN
PROC_AMP_SetGameMode(_Host, "NORMAL");

// mb1 no -> mb2
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Settings023", 0)
THEN
OpenMessageBoxYesNo(_Host, "AMP_Settings024");

// mb2 casual: yes -> casual
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Settings024", 1)
THEN
PROC_AMP_SetGameMode(_Host, "CASUAL");

// mb2 no -> mb3
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Settings024", 0)
THEN
OpenMessageBoxYesNo(_Host, "AMP_Settings025");

// mb3 hard: yes -> hard
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Settings025", 1)
THEN
PROC_AMP_SetGameMode(_Host, "HARD");

// mb3 no -> mb4 stock
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Settings025", 0)
THEN
OpenMessageBoxYesNo(_Host, "AMP_Settings026");

// mb4 stock: yes -> stock
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Settings026", 1)
THEN
PROC_AMP_SetGameMode(_Host, "STOCK");

// mb4 no -> loop back to mb1
IF
MessageBoxYesNoClosed((CHARACTER)_Host, "AMP_Settings026", 0)
THEN
OpenMessageBoxYesNo(_Host, "AMP_Settings023");

// set game mode

// clear the old gamemode fact first so they don't pile up on switch
PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, (STRING)_NewMode)
AND
DB_AMP_GameMode((STRING)_OldGameMode)
THEN
NOT DB_AMP_GameMode(_OldGameMode);

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, (STRING)_Mode)
AND
DB_Players((CHARACTER)_Player)
THEN
// wipe standard statuses for old saves that predate this mode
RemoveStatus(_Player, "AMP_DIFFICULTY_PLAYER_STANDARD", NULL_00000000-0000-0000-0000-000000000000);
RemoveStatus(_Player, "AMP_BOSS_PROGRESS_STANDARD", NULL_00000000-0000-0000-0000-000000000000);

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "CASUAL")
AND
DB_AMP_Difficulty_Mode((STRING)_OldMode)
THEN
NOT DB_AMP_Difficulty_Mode(_OldMode);
DB_AMP_Difficulty_Mode("NONE");
DB_AMP_GameMode("CASUAL");
PROC_AMP_PlayModeEffect(_Host);

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "NORMAL")
AND
DB_AMP_Difficulty_Mode((STRING)_OldMode)
THEN
NOT DB_AMP_Difficulty_Mode(_OldMode);
DB_AMP_Difficulty_Mode("STANDARD");
DB_AMP_GameMode("NORMAL");
PROC_AMP_PlayModeEffect(_Host);

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "HARD")
AND
DB_AMP_Difficulty_Mode((STRING)_OldMode)
THEN
NOT DB_AMP_Difficulty_Mode(_OldMode);
DB_AMP_Difficulty_Mode("IMPOSSIBLE");
DB_AMP_GameMode("HARD");
PROC_AMP_PlayModeEffect(_Host);

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "STOCK")
AND
DB_AMP_Difficulty_Mode((STRING)_OldMode)
THEN
NOT DB_AMP_Difficulty_Mode(_OldMode);
DB_AMP_Difficulty_Mode("NONE");
DB_AMP_GameMode("STOCK");
DB_AMP_BattleInstinct_Disabled(1);
DB_AMP_ItemMoveCost_Disabled(1);
DB_AMP_Wizard_Done(1);
PROC_AMP_PlayModeEffect(_Host);
