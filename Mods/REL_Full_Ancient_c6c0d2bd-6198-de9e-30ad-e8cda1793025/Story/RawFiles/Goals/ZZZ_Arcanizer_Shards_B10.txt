// pack: owner has the shard -> tag the summon with crit dice. no StackId so packs stack.

IF
CharacterJoinedParty((CHARACTER)_Summon)
AND
IsSummon(_Summon, 1)
AND
CharacterGetOwner(_Summon, (CHARACTER)_Owner)
AND
HasActiveStatus(_Owner, "AMP_SHARD_CHAR_PACK", 1)
THEN
ApplyStatus(_Summon, "AMP_SHARD_PACK_CRITDICE", -1.0, 1, _Owner);

// architect: +1 shard slot on the weapon. DB_AMP_Shard_ItemSlotOverride holds the bonus.

// stacking: bump override. must come before the first-apply rule.
IF
StatusApplied((ITEM)_Weapon, "AMP_SHARD_WEAPON_ARCHITECT", _, _)
AND
DB_AMP_Shard_ItemSlotOverride(_Weapon, _OldBonus)
AND
IntegerSum(_OldBonus, 1, _NewBonus)
THEN
NOT DB_AMP_Shard_ItemSlotOverride(_Weapon, _OldBonus);
DB_AMP_Shard_ItemSlotOverride(_Weapon, _NewBonus);

// first apply on melee weapon: record override
IF
StatusApplied((ITEM)_Weapon, "AMP_SHARD_WEAPON_ARCHITECT", _, _)
AND
NOT DB_AMP_Shard_ItemSlotOverride(_Weapon, _)
THEN
DB_AMP_Shard_ItemSlotOverride(_Weapon, 1);

// marker removed: decrement override
IF
StatusRemoved((ITEM)_Weapon, "AMP_SHARD_WEAPON_ARCHITECT", _, _)
AND
DB_AMP_Shard_ItemSlotOverride(_Weapon, _OldBonus)
AND
_OldBonus > 1
AND
IntegerSubtract(_OldBonus, 1, _NewBonus)
THEN
NOT DB_AMP_Shard_ItemSlotOverride(_Weapon, _OldBonus);
DB_AMP_Shard_ItemSlotOverride(_Weapon, _NewBonus);

IF
StatusRemoved((ITEM)_Weapon, "AMP_SHARD_WEAPON_ARCHITECT", _, _)
AND
DB_AMP_Shard_ItemSlotOverride(_Weapon, 1)
THEN
NOT DB_AMP_Shard_ItemSlotOverride(_Weapon, 1);

// CanApply override: item with a slot bonus allows up to limit+bonus shards
QRY
QRY_AMP_Shard_CanApply((ITEM)_Item)
AND
ItemGetGoldValue(_Item, _Value)
AND
DB_AMP_Shard_PriceRange(_Rarity, _Min, _Max)
AND
_Value >= _Min
AND
_Value <= _Max
AND
DB_AMP_Shard_SlotLimit(_Rarity, _Limit)
AND
DB_AMP_Shard_TotalCount(_Item, _Current)
AND
DB_AMP_Shard_ItemSlotOverride(_Item, _Bonus)
AND
IntegerSum(_Limit, _Bonus, _EffectiveLimit)
AND
_Current < _EffectiveLimit
THEN
DB_NOOP(1);
