KBSECTION

// existing-save hotfix: old main goal applied school markers at 6.0 duration
// (1 combat turn) so they expired before the turn-tick cleanup. re-apply the
// same school at infinite duration while the old version is loaded (no FixV2 marker).

IF
StatusApplied((CHARACTER)_Char, (STRING)_Status, _, _)
AND
NOT DB_AMP_UL_FixV2(1)
AND
HasActiveStatus(_Char, "AMP_ARCANE_RECLAIM_TECH", 1)
AND
DB_AMP_UL_School(_, _Status)
AND
NOT DB_AMP_UL_HotfixDone(_Char, _Status)
THEN
DB_AMP_UL_HotfixDone(_Char, _Status);
ApplyStatus(_Char, _Status, -1.0, 1, _Char);

IF
TurnStarted((CHARACTER)_Char)
AND
DB_AMP_UL_HotfixDone(_Char, (STRING)_S)
THEN
NOT DB_AMP_UL_HotfixDone(_Char, _S);

IF
StatusRemoved((CHARACTER)_Char, "AMP_ARCANE_RECLAIM_TECH", _, _)
AND
DB_AMP_UL_HotfixDone(_Char, (STRING)_S)
THEN
NOT DB_AMP_UL_HotfixDone(_Char, _S);

EXITSECTION
ENDEXITSECTION
