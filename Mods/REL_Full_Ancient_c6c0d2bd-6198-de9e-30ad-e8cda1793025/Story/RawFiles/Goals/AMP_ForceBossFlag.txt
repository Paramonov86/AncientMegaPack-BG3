// At combat start, force IsBoss on any roster boss that is hostile, not a party member,
// and currently missing the flag. Hostility/party are read now, so recruited Minthara and
// the freed Nightsong ally are correctly skipped.
IF
EnteredCombat((CHARACTER)_C, _)
AND
DB_AMP_ForceBoss(_C)
AND
IsBoss(_C, 0)
AND
GetHostCharacter((CHARACTER)_Host)
AND
IsEnemy(_C, _Host, 1)
AND
IsPartyMember(_C, 1, 0)
THEN
SetIsBoss(_C, 1);

EXITSECTION
