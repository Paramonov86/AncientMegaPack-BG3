// track who used the station
IF
UseStarted((CHARACTER)_Character, (ITEM)_Station)
AND
GetTemplate(_Station, (ITEMROOT)_StationTemplate)
AND
DB_AMP_RandomStation(_StationTemplate)
THEN
// seed crystal costs on old saves
PROC_AMP_InitializeCrystalCosts();
// clear stuck flags
NOT DB_AMP_Matrix_JustDepleted(_Station);
SysClear("DB_AMP_RandomStation_LastUser", 1);
DB_AMP_RandomStation_LastUser(_Character);
// track this instance for status restore after long rest
PROC_AMP_TrackStationInstance(_Station);

// item dropped into the station
IF
AddedTo(_Item, _Container, _)
AND
DB_AMP_RandomStation((ITEMROOT)_StationTemplate)
AND
GetTemplate(_Container, _StationTemplate)
AND
NOT DB_AMP_RandomProcessing(_Item)
THEN
// mark processing so we don't loop
DB_AMP_RandomProcessing(_Item);
PROC_AMP_CheckItemType((ITEM)_Item, (ITEM)_Container);

// matrix vs crystal

// resonance matrix -> queue it
PROC
PROC_AMP_CheckItemType((ITEM)_Item, (ITEM)_Container)
AND
GetTemplate(_Item, (ITEMROOT)_ItemTemplate)
AND
_ItemTemplate == Resonance_Matrix_2c8dfeb6-c86e-4595-b9e9-080aa87bd0b7
THEN
PROC_AMP_QueueMatrix((ITEM)_Item, (ITEM)_Container);

// anything else (crystals)
PROC
PROC_AMP_CheckItemType((ITEM)_Item, (ITEM)_Container)
AND
GetTemplate(_Item, (ITEMROOT)_ItemTemplate)
AND
NOT _ItemTemplate == Resonance_Matrix_2c8dfeb6-c86e-4595-b9e9-080aa87bd0b7
THEN
PROC_AMP_CheckAndProcessItem((ITEM)_Item, (ITEM)_Container);

// matrix queue - process one at a time to avoid race conditions

// queue a matrix
PROC
PROC_AMP_QueueMatrix((ITEM)_Matrix, (ITEM)_Station)
THEN
DB_AMP_MatrixQueue(_Matrix, _Station);
PROC_AMP_TryProcessNextMatrix();

// process next if idle
PROC
PROC_AMP_TryProcessNextMatrix()
AND
NOT DB_AMP_MatrixProcessingActive(_)
AND
DB_AMP_MatrixQueue((ITEM)_Matrix, (ITEM)_Station)
THEN
DB_AMP_MatrixProcessingActive(1);
NOT DB_AMP_MatrixQueue(_Matrix, _Station);
PROC_AMP_InstallMatrix((ITEM)_Matrix, (ITEM)_Station);
TimerLaunch("AMP_ProcessNextMatrix", 200);

// empty queue -> nothing
PROC
PROC_AMP_TryProcessNextMatrix()
AND
NOT DB_AMP_MatrixQueue(_, _)
THEN
DB_NOOP(1);

// timer -> clear flag, process next
IF
TimerFinished("AMP_ProcessNextMatrix")
THEN
SysClear("DB_AMP_MatrixProcessingActive", 1);
PROC_AMP_TryProcessNextMatrix();

THEN

AND

AND

AND
NOT IsInventoryEmpty(_Container, 1)

NOT DB_AMP_ContainerSoundPlayed((ITEM)_Container);

// delete the reward container if empty

AND

AND
DB_AMP_RandomRecipe(_, _Template)
AND
IsInventoryEmpty(_Container, 1)
AND
NOT DB_AMP_RandomProcessing(_Container)
THEN

ObjectTimerFinished(_Container, "AMP_DeleteEmptyRewardContainer")
AND
GetTemplate(_Container, (ITEMROOT)_Template)
AND
DB_AMP_RandomRecipe(_, _Template)
AND
DB_AMP_RandomProcessing(_Container)
AND
IsInventoryEmpty(_Container, 1)
THEN
NOT DB_AMP_RandomProcessing((ITEM)_Container);
RequestDelete((ITEM)_Container);

ObjectTimerFinished(_Container, "AMP_DeleteEmptyRewardContainer")
AND
DB_AMP_RandomProcessing(_Container)
AND
NOT IsInventoryEmpty(_Container, 1)
THEN
NOT DB_AMP_RandomProcessing((ITEM)_Container);

AND

THEN
DB_AMP_TrackedStation(_Station);

PROC_AMP_TrackStationInstance((ITEM)_Station)
AND
DB_AMP_TrackedStation(_Station)
THEN
DB_NOOP(1);

// long rest -> restore matrix status on tracked stations that still have charges
IF
LongRestFinished()
AND
DB_AMP_Matrix_Charges(_Charges)
AND
_Charges > 0
AND
DB_AMP_TrackedStation((ITEM)_Station)
AND
HasActiveStatus(_Station, "AMP_MATRIX_ACTIVE", 0)
AND
GetHostCharacter((CHARACTER)_Player)
THEN
ApplyStatus(_Station, "AMP_MATRIX_ACTIVE", -1.0, 1, _Player);
