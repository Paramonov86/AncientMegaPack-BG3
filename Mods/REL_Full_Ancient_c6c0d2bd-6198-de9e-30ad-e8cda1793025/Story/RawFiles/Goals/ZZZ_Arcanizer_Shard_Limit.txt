KBSECTION

// can a shard be applied? (limit by rarity)

// first shard, no total count yet = allowed.
// also require no per-shard rows, else legacy items bypass the limit as "fresh".
QRY
QRY_AMP_Shard_CanApply((ITEM)_Item)
AND
NOT DB_AMP_Shard_TotalCount(_Item, _)
AND
NOT DB_AMP_Shard_Count(_Item, _, _)
THEN
DB_NOOP(1);

// later shards - check limit by price
QRY
QRY_AMP_Shard_CanApply((ITEM)_Item)
AND
ItemGetGoldValue(_Item, _Value)
AND
DB_AMP_Shard_PriceRange(_Rarity, _Min, _Max)
AND
_Value >= _Min
AND
_Value <= _Max
AND
DB_AMP_Shard_SlotLimit(_Rarity, _Limit)
AND
DB_AMP_Shard_TotalCount(_Item, _Current)
AND
_Current < _Limit
THEN
DB_NOOP(1);

// inc/dec total count

PROC
PROC_AMP_Shard_IncTotal((ITEM)_Item)
AND
DB_AMP_Shard_TotalCount(_Item, _Old)
AND
IntegerSum(_Old, 1, _New)
THEN
NOT DB_AMP_Shard_TotalCount(_Item, _Old);
DB_AMP_Shard_TotalCount(_Item, _New);

PROC
PROC_AMP_Shard_IncTotal((ITEM)_Item)
AND
NOT DB_AMP_Shard_TotalCount(_Item, _)
THEN
DB_AMP_Shard_TotalCount(_Item, 1);

PROC
PROC_AMP_Shard_DecTotal((ITEM)_Item)
AND
DB_AMP_Shard_TotalCount(_Item, _Old)
AND
_Old > 1
AND
IntegerSubtract(_Old, 1, _New)
THEN
NOT DB_AMP_Shard_TotalCount(_Item, _Old);
DB_AMP_Shard_TotalCount(_Item, _New);

PROC
PROC_AMP_Shard_DecTotal((ITEM)_Item)
AND
DB_AMP_Shard_TotalCount(_Item, 1)
THEN
NOT DB_AMP_Shard_TotalCount(_Item, 1);

AND
HasActiveStatus(_Instance, "TOT_SHOP_ITEM", 0)
AND
