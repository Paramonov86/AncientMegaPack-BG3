// corpse loot via movealllootableitemsto. two triggers, one proc:
// post-combat (died -> queued -> combatended+2s) and environmental (marker on corpse).
// engine handles npc-owner protection.

// corpse loot proc. vfx deferred to removedfrom so it only fires if something moves.
PROC
PROC_AMP_AL_LootCorpse((CHARACTER)_Corpse, (CHARACTER)_Player)
AND
IsDead(_Corpse, 1)
AND
Exists(_Corpse, 1)
AND
NOT DB_Players(_Corpse)
AND
NOT DB_OwnedCorpse(_Corpse, _)
AND
IsPartyMember(_Corpse, 1, 0)
AND
IsAlly(_Corpse, _Player, 0)
AND
GetPosition(_Corpse, (REAL)_X, (REAL)_Y, (REAL)_Z)
THEN
DebugText(_Player, "[AL] Corpse loot attempt");
DB_AMP_AL_CorpseVFXPos(_Corpse, _X, _Y, _Z);
DB_AMP_AL_CampSendCorpse(_Corpse, _Player);
MoveAllLootableItemsTo(_Corpse, _Player, 1, 1, 0, 0);
ObjectTimerLaunch(_Corpse, "AMP_AL_CorpseVFXCleanup", 1000);

// vfx on first item leaving the corpse
IF
RemovedFrom(_Item, (CHARACTER)_Corpse)
AND
DB_AMP_AL_CorpseVFXPos(_Corpse, _X, _Y, _Z)
THEN
PlayEffectAtPosition(AMP_Autoloot_Marker_af1b6860-8b91-5f0a-9553-831bc6c8e10d, _X, _Y, _Z, 1.0);
NOT DB_AMP_AL_CorpseVFXPos(_Corpse, _X, _Y, _Z);

// nothing moved (empty corpse): drop stale entry
IF
ObjectTimerFinished((CHARACTER)_Corpse, "AMP_AL_CorpseVFXCleanup")
AND
DB_AMP_AL_CorpseVFXPos(_Corpse, _X, _Y, _Z)
THEN
NOT DB_AMP_AL_CorpseVFXPos(_Corpse, _X, _Y, _Z);

// died in combat -> queue for post-combat
IF
Died((CHARACTER)_Corpse)
AND
NOT DB_AMP_AutoLoot_Disabled(1)
AND
NOT IsPlayer(_Corpse, 1)
AND
NOT DB_Players(_Corpse)
AND
NOT DB_OwnedCorpse(_Corpse, _)
AND
IsPartyMember(_Corpse, 1, 0)
AND
QRY_AMP_AL_AnyPlayerHasAura()
THEN
DB_AMP_AL_PendingCorpse(_Corpse);
DebugText(_Corpse, "[AL] Died -> queued");

// post-combat: process pending corpses
IF
CombatEnded(_)
AND
NOT DB_AMP_AutoLoot_Disabled(1)
AND
QRY_AMP_AL_AnyPlayerHasAura()
THEN
TimerLaunch("AMP_AL_PostCombatLoot", 2000);

// close (<=15m) -> loot + drop from queue
IF
TimerFinished("AMP_AL_PostCombatLoot")
AND
GetHostCharacter((CHARACTER)_Host)
AND
IsCharacter(_Host, 1)
AND
IsInCombat(_Host, 0)
AND
NOT DB_AMP_AutoLoot_Disabled(1)
AND
DB_AMP_AL_PendingCorpse((CHARACTER)_Corpse)
AND
GetDistanceTo(_Host, _Corpse, (REAL)_Dist)
AND
_Dist <= 15.0
THEN
NOT DB_AMP_AL_PendingCorpse(_Corpse);
PROC_AMP_AL_LootCorpse(_Corpse, _Host);

// far (>15m) -> drop from queue, aura grabs it when player approaches
IF
TimerFinished("AMP_AL_PostCombatLoot")
AND
GetHostCharacter((CHARACTER)_Host)
AND
IsCharacter(_Host, 1)
AND
NOT DB_AMP_AutoLoot_Disabled(1)
AND
DB_AMP_AL_PendingCorpse((CHARACTER)_Corpse)
AND
GetDistanceTo(_Host, _Corpse, (REAL)_Dist)
AND
_Dist > 15.0
THEN
NOT DB_AMP_AL_PendingCorpse(_Corpse);

// drop stale pending entries
IF
TimerFinished("AMP_AL_PostCombatLoot")
AND
DB_AMP_AL_PendingCorpse((CHARACTER)_Corpse)
AND
Exists(_Corpse, 0)
THEN
NOT DB_AMP_AL_PendingCorpse(_Corpse);
