// spawn reward - deferredspawn with a unique tag
PROC
PROC_AMP_VoidCapsule_SpawnReward((CHARACTER)_Owner, (STRING)_Context, (ITEMROOT)_Template, (ITEM)_Capsule)
AND
DB_AMP_VoidCapsule_DeferredCounter((INTEGER)_N)
AND
IntegerSum(_N, 1, (INTEGER)_Tag)
THEN
NOT DB_AMP_VoidCapsule_DeferredCounter(_N);
DB_AMP_VoidCapsule_DeferredCounter(_Tag);
DB_AMP_VoidCapsule_DeferredSpawn(_Tag, _Owner, _Context, _Template, _Capsule);

// seedreset done callback - spawns the real reward when the tag matches
PROC
PROC_AMP_SeedReset_Done((CHARACTER)_Owner, (ITEMROOT)_Template)
AND
DB_AMP_VoidCapsule_SeedReset_TagPending((INTEGER)_Tag, _Owner, _Template)
AND
DB_AMP_VoidCapsule_DeferredSpawn(_Tag, _Owner, (STRING)_Context, _Template, (ITEM)_Capsule)
THEN
NOT DB_AMP_VoidCapsule_SeedReset_TagPending(_Tag, _Owner, _Template);
NOT DB_AMP_VoidCapsule_DeferredSpawn(_Tag, _Owner, _Context, _Template, _Capsule);
TemplateAddTo(_Template, _Owner, 1, 0);
DB_AMP_VoidCapsule_PendingReward(_Owner, _Context, _Template, _Capsule);

// no tag pending - seedreset fired without spawnreward
PROC
PROC_AMP_SeedReset_Done((CHARACTER)_Owner, (ITEMROOT)_Template)
AND
NOT DB_AMP_VoidCapsule_SeedReset_TagPending(_, _Owner, _Template)
THEN
DB_NOOP(1);

// 19. reward ux - usestarted (player opens reward cube)
IF
UseStarted((CHARACTER)_Player, (ITEM)_Container)
AND
GetTemplate(_Container, (ITEMROOT)_Template)
AND
DB_AMP_VoidCapsule_Reward(_, _, _Template)
AND
NOT DB_AMP_VoidCapsule_ContainerOpened(_Container)
THEN
DB_AMP_VoidCapsule_ContainerOpened(_Container);
PlaySound(_Player, "SCL_HarperScouts_ShadowCreatures_ImpactRiser_B");
ObjectTimerLaunch(_Container, "AMP_VoidCapsule_ForceDelete", 5000, 0);

// 20. reward ux - forcedelete timer finished + empty
IF
ObjectTimerFinished((ITEM)_Container, "AMP_VoidCapsule_ForceDelete")
AND
IsInventoryEmpty(_Container, 1)
THEN
NOT DB_AMP_VoidCapsule_ContainerOpened(_Container);
RequestDelete(_Container);

// forcedelete timer finished but container still has items - leave it alone
IF
ObjectTimerFinished((ITEM)_Container, "AMP_VoidCapsule_ForceDelete")
AND
NOT IsInventoryEmpty(_Container, 1)
THEN
DB_NOOP(1);

// 21. reward ux - auto-delete emptied reward container
IF
RemovedFrom((ITEM)_Item, (ITEM)_Container)
AND
GetTemplate(_Container, (ITEMROOT)_Template)
AND
DB_AMP_VoidCapsule_Reward(_, _, _Template)
AND
IsInventoryEmpty(_Container, 1)
AND
NOT DB_AMP_VoidCapsule_Deleting(_Container)
THEN
DB_AMP_VoidCapsule_Deleting(_Container);
ObjectTimerLaunch(_Container, "AMP_VoidCapsule_DeleteEmpty", 200, 0);

// deleteempty timer finished + still empty - drop container
IF
ObjectTimerFinished((ITEM)_Container, "AMP_VoidCapsule_DeleteEmpty")
AND
IsInventoryEmpty(_Container, 1)
THEN
NOT DB_AMP_VoidCapsule_Deleting(_Container);
NOT DB_AMP_VoidCapsule_ContainerOpened(_Container);
RequestDelete(_Container);

// deleteempty timer finished + not empty - cancel deletion
IF
ObjectTimerFinished((ITEM)_Container, "AMP_VoidCapsule_DeleteEmpty")
AND
NOT IsInventoryEmpty(_Container, 1)
THEN
NOT DB_AMP_VoidCapsule_Deleting(_Container);

// 22. reward ux - auto-extract from any voidcapsule reward container
// applies to coin rewards, salvage shard cubes, and reroll respawns.
// removedfrom fires per item, so reroll-marking (status apply on removedfrom)
// still runs for reroll-context cubes. empty cube then auto-deletes.
// decoy cubes (seedreset) are excluded so their treasuretable loot is not
// ejected to the player - decoys must disappear silently.
IF
AddedTo((ITEM)_Item, (ITEM)_Container, _)
AND
GetTemplate(_Container, (ITEMROOT)_Template)
AND
DB_AMP_VoidCapsule_Reward(_, _, _Template)
AND
NOT DB_AMP_SeedReset_DecoyCube(_Container)
THEN
ObjectTimerLaunch(_Item, "AMP_VoidCapsule_Eject", 100, 0);
