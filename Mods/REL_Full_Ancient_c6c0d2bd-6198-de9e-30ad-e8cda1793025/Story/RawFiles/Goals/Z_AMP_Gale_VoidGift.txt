// gale rewards the player with random void capsules after his personal story
// beats. hard mode only; only while gale is a recruited companion (not the
// player's avatar) and in the active controllable party.

// shared grant procedures

// generic: messagebox + one random capsule. runs for every grant.
PROC
PROC_AMP_GaleVoidGift_Grant((CHARACTER)_Host, (INTEGER)_Extra)
THEN
OpenMessageBox(_Host, "AMP_Gale_VoidGift");
PROC_AMP_GaleVoidGift_Capsule(_Host);

// literal-match overload: a second capsule when the 0..1 roll came up 1.
PROC
PROC_AMP_GaleVoidGift_Grant((CHARACTER)_Host, 1)
THEN
PROC_AMP_GaleVoidGift_Capsule(_Host);

// pick one random capsule type from the 22-entry registry and add it.
PROC
PROC_AMP_GaleVoidGift_Capsule((CHARACTER)_Host)
AND
QRY_GetRandom("DB_AMP_VoidCapsule_Container", 2, "DB_AMP_GaleVoidGift_RandCap")
AND
DB_AMP_GaleVoidGift_RandCap((ITEMROOT)_Template, (STRING)_Type)
THEN
NOT DB_AMP_GaleVoidGift_RandCap(_Template, _Type);
TemplateAddTo(_Template, _Host, 1, 1);

// m1: gale reveals the netherese orb (learnedbackstory)
IF
DialogEnded(Gale_InParty2_6beb1b10-845f-49fa-6d6d-f425eaa42574, _)
AND
NOT DB_AMP_GaleVoidGift_Done("M1")
AND
DB_AMP_GameMode("HARD")
AND
DB_GlobalFlag((FLAG)ORI_Gale_Knows_LearnedBackstory_0a9731cf-4769-249f-818d-b4c6e542083d)
AND
IsPartyMember((CHARACTER)S_Player_Gale_ad9af97d-75da-406a-ae13-7071c563f604, 0, 1)
AND
NOT DB_Avatars((CHARACTER)S_Player_Gale_ad9af97d-75da-406a-ae13-7071c563f604)
AND
GetHostCharacter(_Host)
AND
Random(2, _Extra)
THEN
DB_AMP_GaleVoidGift_Done("M1");
PROC_AMP_GaleVoidGift_Grant(_Host, _Extra);
