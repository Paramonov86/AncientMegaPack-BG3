Version 1
SubGoalCombiner SGC_AND
INITSECTION

DB_AMP_Shard_CountFix_Enabled(1);

KBSECTION

// engine refires statusapplied for statuses on worn items when a save loads.
// the core transfer counter treats every refire as a fresh shard application,
// so the per-marker count inflates by one on every load and the reclamation
// shard then returns that many copies. a real shard application always arrives
// with a player causee (templateusestarted handlers apply the tech status with
// the character as causee and the chain keeps it); a load refire does not.
// compensate: take the increment back whenever the causee is not a player.
// order-safe vs the core rules: net result is identical whichever fires first.

IF
StatusApplied((ITEM)_Item, (STRING)_ItemStatus, (CHARACTER)_Causee, _)
AND
DB_AMP_Shard_CountFix_Enabled(1)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _)
AND
NOT DB_Players(_Causee)
AND
DB_AMP_Shard_Count(_Item, _ItemStatus, (INTEGER)_Count)
AND
_Count > 1
AND
IntegerSubtract(_Count, 1, (INTEGER)_New)
THEN
NOT DB_AMP_Shard_Count(_Item, _ItemStatus, _Count);
DB_AMP_Shard_Count(_Item, _ItemStatus, _New);

IF
StatusApplied((ITEM)_Item, (STRING)_ItemStatus, (CHARACTER)_Causee, _)
AND
DB_AMP_Shard_CountFix_Enabled(1)
AND
DB_AMP_Shard_Transfer(_ItemStatus, _)
AND
NOT DB_Players(_Causee)
AND
DB_AMP_Shard_Count(_Item, _ItemStatus, 1)
THEN
NOT DB_AMP_Shard_Count(_Item, _ItemStatus, 1);

EXITSECTION
ENDEXITSECTION
