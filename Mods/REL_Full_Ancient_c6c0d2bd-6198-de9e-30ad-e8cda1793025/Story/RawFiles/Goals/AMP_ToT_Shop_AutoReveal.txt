KBSECTION

// shop containers respawn INVISIBLE on level change, so the shop vanished even after
// the player turned it on. on level start, if already activated, strip INVISIBLE.
// no stock changes, restock handles first use.

IF
LevelGameplayStarted(_, _)
AND
DB_ToT_Shop_Enabled(1)
AND
DB_ToT_Shop_Activated(1)
AND
DB_ToT_Shop((STRING)_ShopID, (ITEM)_Container)
THEN
RemoveStatus(_Container, "INVISIBLE", _Container);

EXITSECTION
ENDEXITSECTION
