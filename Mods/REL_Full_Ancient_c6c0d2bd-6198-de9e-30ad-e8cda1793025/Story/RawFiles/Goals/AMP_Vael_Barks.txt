SubGoalCombiner SGC_AND
INITSECTION

KBSECTION
// vael combat barks (mirror doppelganger). it copies a player on combat enter so
// the template stops matching - gate on the chase marker and tag it after it takes
// a form so the bark dialog can route to it.

IF
StatusApplied((CHARACTER)_Mob, "AMP_MIRROR_INFORM", _, _)
AND
DB_AMP_MirrorChase_Mob(_Mob, AMP_ULTRA_Summon_05_f8560efd-1d3c-4dcc-8b4e-b36f2cc79684)
AND
NOT DB_AMP_Vael_Ready(_Mob)
THEN
DB_AMP_Vael_Ready(_Mob);
SetTag(_Mob, (TAG)AMP_TAG_BARK_VAEL_433de06d-19be-4486-98ce-5a982c55a0f3);

IF
TurnStarted((CHARACTER)_Mob)
AND
DB_AMP_Vael_Ready(_Mob)
AND
NOT DB_AMP_Vael_Greeted(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Vael_Greeting_1fd6dbfc-d737-4c1a-8245-c3ac764bf866, _Mob)
THEN
DB_AMP_Vael_Greeted(_Mob);
DB_AMP_Vael_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Vael_BarkUnlock", 5000);

IF
UsingSpell((CHARACTER)_Mob, _, _, _, _)
AND
DB_AMP_Vael_Ready(_Mob)
AND
IsDead(_Mob, 0)
AND
NOT DB_AMP_Vael_BarkLock(_Mob)
AND
Random(3, (INTEGER)_Roll)
AND
_Roll == 0
THEN
DB_AMP_Vael_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Vael_BarkDelay", 1500);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Vael_BarkDelay")
AND
DB_AMP_Vael_BarkLock(_Mob)
AND
IsDead(_Mob, 0)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Vael_SpellCast_c44e871b-b717-407b-84ae-870077009589, _Mob)
THEN
RealtimeObjectTimerLaunch(_Mob, "AMP_Vael_BarkUnlock", 10000);

IF
ObjectTimerFinished((CHARACTER)_Mob, "AMP_Vael_BarkUnlock")
AND
DB_AMP_Vael_BarkLock(_Mob)
THEN
NOT DB_AMP_Vael_BarkLock(_Mob);

IF
HitpointsChanged((CHARACTER)_Mob, (REAL)_Pct)
AND
DB_AMP_Vael_Ready(_Mob)
AND
_Pct <= 35.0
AND
NOT DB_AMP_Vael_LowHP(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Vael_LowHP_8970aabe-15fb-4b57-8903-a3129989358a, _Mob)
THEN
DB_AMP_Vael_LowHP(_Mob);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Vael_Ready(_Mob)
AND
QRY_StartDialog((DIALOGRESOURCE)AMP_Bark_Vael_Death_524673a8-8e89-491d-935c-f5505994a06a, _Mob)
THEN
DB_NOOP(1);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Vael_Ready(_Mob)
THEN
NOT DB_AMP_Vael_Ready(_Mob);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Vael_Greeted(_Mob)
THEN
NOT DB_AMP_Vael_Greeted(_Mob);
DB_AMP_Vael_BarkLock(_Mob);
RealtimeObjectTimerLaunch(_Mob, "AMP_Vael_BarkUnlock", 5000);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Vael_BarkLock(_Mob)
THEN
NOT DB_AMP_Vael_BarkLock(_Mob);

IF
Died((CHARACTER)_Mob)
AND
DB_AMP_Vael_LowHP(_Mob)
THEN
NOT DB_AMP_Vael_LowHP(_Mob);

EXITSECTION

ENDEXITSECTION
