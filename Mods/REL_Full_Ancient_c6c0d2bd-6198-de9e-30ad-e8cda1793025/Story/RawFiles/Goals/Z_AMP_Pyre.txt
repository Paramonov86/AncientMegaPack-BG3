INITSECTION
DB_AMP_Pyre_Disabled(1);

KBSECTION
// pyre: escalating bonfire keyed to combat tempo. points accrue per boss-fight
// from damage taken/kills/cc, decay each round, doused by acid/radiant. hidden
// points map to 10 gradation statuses on every live hostile.

// active for current difficulty + toggle
QRY
QRY_AMP_Pyre_Active()
AND
DB_AMP_Difficulty_Mode("IMPOSSIBLE")
THEN
DB_NOOP(1);

QRY
QRY_AMP_Pyre_Active()
AND
DB_AMP_Difficulty_Mode("STANDARD")
AND
NOT DB_AMP_Pyre_Disabled(1)
THEN
DB_NOOP(1);

// hostile to the players
QRY
QRY_AMP_Pyre_IsHostile((CHARACTER)_Char)
AND
DB_Players((CHARACTER)_Player)
AND
IsEnemy(_Player, _Char, 1)
THEN
DB_NOOP(1);

// combat membership and arming

// record every hostile entering any combat
IF
EnteredCombat((CHARACTER)_Char, (GUIDSTRING)_CombatID)
AND
QRY_AMP_Pyre_Active()
AND
QRY_AMP_Pyre_IsHostile(_Char)
THEN
DB_AMP_Pyre_Combatant(_Char, _CombatID);

// boss in the fight arms it
IF
EnteredCombat((CHARACTER)_Char, (GUIDSTRING)_CombatID)
AND
IsBoss(_Char, 1)
AND
NOT DB_AMP_Pyre_BossCombat(_CombatID)
THEN
DB_AMP_Pyre_BossCombat(_CombatID);
PROC_AMP_Pyre_ArmCombat(_CombatID);

// hostile joining an armed fight: track + catch up
IF
EnteredCombat((CHARACTER)_Char, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
AND
QRY_AMP_Pyre_IsHostile(_Char)
THEN
ApplyStatus(_Char, "AMP_PYRE_TRACKER", -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);
PROC_AMP_Pyre_CatchUpOne(_Char, _CombatID);

// tracker to all hostiles already present when boss appears
PROC
PROC_AMP_Pyre_ArmCombat((GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_Combatant((CHARACTER)_Char, _CombatID)
AND
IsDead(_Char, 0)
AND
QRY_AMP_Pyre_IsHostile(_Char)
THEN
ApplyStatus(_Char, "AMP_PYRE_TRACKER", -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);

// late joiner only: give the new hostile the current grade, don't re-blast everyone
PROC
PROC_AMP_Pyre_CatchUpOne((CHARACTER)_Char, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_Grade(_CombatID, (INTEGER)_G)
AND
_G > 0
THEN
PROC_AMP_Pyre_ApplyGradeOne(_Char, _G);

// point accrual (markers from tracker passives + kill events)

// enemy took damage: +2 (tempo: damage now weighs as much as a cc landing)
IF
StatusApplied((CHARACTER)_Char, "AMP_PYRE_MARK_DMG", _, _)
AND
DB_AMP_Pyre_Combatant(_Char, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
THEN
PROC_AMP_Pyre_AddPoints(_CombatID, 2);

// acid/radiant taken: -6 (net -4 with the +2 damage tick; plus the per-round douse-kick)
IF
StatusApplied((CHARACTER)_Char, "AMP_PYRE_MARK_DOUSE", _, _)
AND
DB_AMP_Pyre_Combatant(_Char, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
THEN
PROC_AMP_Pyre_AddPoints(_CombatID, -6);

// douse kick rearms every round: clear the guard at round start
IF
CombatRoundStarted((GUIDSTRING)_CombatID, _)
AND
DB_AMP_Pyre_FirstDouse(_CombatID)
THEN
NOT DB_AMP_Pyre_FirstDouse(_CombatID);

// first douse this round: drop a grade now so it feels responsive, not at round tick
IF
StatusApplied((CHARACTER)_Char, "AMP_PYRE_MARK_DOUSE", _, _)
AND
DB_AMP_Pyre_Combatant(_Char, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
AND
NOT DB_AMP_Pyre_FirstDouse(_CombatID)
THEN
DB_AMP_Pyre_FirstDouse(_CombatID);
PROC_AMP_Pyre_DouseKick(_CombatID);

// cc landed on an enemy: +2
IF
StatusApplied((CHARACTER)_Char, "AMP_PYRE_MARK_CC", _, _)
AND
DB_AMP_Pyre_Combatant(_Char, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
THEN
PROC_AMP_Pyre_AddPoints(_CombatID, 2);

// enemy healed: -24 flat (healing an enemy hard-douses the fire; 3x the old dock)
IF
StatusApplied((CHARACTER)_Char, "AMP_PYRE_MARK_HEAL", _, _)
AND
DB_AMP_Pyre_Combatant(_Char, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
THEN
PROC_AMP_Pyre_AddPoints(_CombatID, -24);

// enemy failed a con save: -1
IF
StatusApplied((CHARACTER)_Char, "AMP_PYRE_MARK_CON", _, _)
AND
DB_AMP_Pyre_Combatant(_Char, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
THEN
PROC_AMP_Pyre_AddPoints(_CombatID, -1);

// any hostile dies in an armed fight: +15
IF
Died((CHARACTER)_Dead)
AND
DB_AMP_Pyre_Combatant(_Dead, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
AND
QRY_AMP_Pyre_IsHostile(_Dead)
THEN
PROC_AMP_Pyre_AddPoints(_CombatID, 15);

// dead one was a boss: +65 more (total +80)
IF
Died((CHARACTER)_Dead)
AND
DB_AMP_Pyre_Combatant(_Dead, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
AND
IsBoss(_Dead, 1)
THEN
PROC_AMP_Pyre_AddPoints(_CombatID, 65);

// on death recompute grade now on raw points, so a fast kill flares instantly
// instead of waiting for the round-start halving. runs after the accrual above
// so it sees this kill's points. round tick still settles afterward.
IF
Died((CHARACTER)_Dead)
AND
DB_AMP_Pyre_Combatant(_Dead, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
AND
QRY_AMP_Pyre_IsHostile(_Dead)
AND
NOT DB_AMP_Pyre_DeathRecomputed(_CombatID)
AND
DB_AMP_Pyre_Points(_CombatID, (INTEGER)_N)
THEN
DB_AMP_Pyre_DeathRecomputed(_CombatID);
PROC_AMP_Pyre_Recompute(_CombatID, _N);

// death flare fires once per turn: clear the guard at each turn start
IF
TurnStarted((GUIDSTRING)_Obj)
AND
CombatGetGuidFor(_Obj, (GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_DeathRecomputed(_CombatID)
THEN
NOT DB_AMP_Pyre_DeathRecomputed(_CombatID);

// round tick: keep 60% of points then grade once. grade only moves here, so the
// fire steps once per round. slow down and the decay outruns accrual -> it cools,
// but slower now (was halve/50%) so the hell lingers longer between turns.
IF
CombatRoundStarted((GUIDSTRING)_CombatID, (INTEGER)_Round)
AND
DB_AMP_Pyre_BossCombat(_CombatID)
AND
QRY_AMP_Pyre_Active()
AND
DB_AMP_Pyre_Points(_CombatID, (INTEGER)_N)
THEN
PROC_AMP_Pyre_RoundTick(_CombatID, _N);

PROC
PROC_AMP_Pyre_RoundTick((GUIDSTRING)_CombatID, (INTEGER)_N)
AND
IntegerProduct(_N, 3, (INTEGER)_X)
AND
IntegerDivide(_X, 5, (INTEGER)_Kept)
THEN
PROC_AMP_Pyre_Write(_CombatID, _Kept);
PROC_AMP_Pyre_Recompute(_CombatID, _Kept);

// knock the live grade down one step now (no-op below grade 1)
PROC
PROC_AMP_Pyre_DouseKick((GUIDSTRING)_CombatID)
AND
DB_AMP_Pyre_Grade(_CombatID, (INTEGER)_G)
AND
_G > 0
AND
IntegerSubtract(_G, 1, (INTEGER)_New)
THEN
PROC_AMP_Pyre_SetGrade(_CombatID, _New);

// point storage + clamp at 0 (grade recomputed once per round, not per change)
PROC
PROC_AMP_Pyre_AddPoints((GUIDSTRING)_CombatID, (INTEGER)_Delta)
AND
DB_AMP_Pyre_Points(_CombatID, (INTEGER)_Old)
AND
IntegerSum(_Old, _Delta, (INTEGER)_Raw)
THEN
PROC_AMP_Pyre_Commit(_CombatID, _Raw);

PROC
PROC_AMP_Pyre_AddPoints((GUIDSTRING)_CombatID, (INTEGER)_Delta)
AND
NOT DB_AMP_Pyre_Points(_CombatID, _)
AND
IntegerSum(0, _Delta, (INTEGER)_Raw)
THEN
PROC_AMP_Pyre_Commit(_CombatID, _Raw);

PROC
PROC_AMP_Pyre_Commit((GUIDSTRING)_CombatID, (INTEGER)_Raw)
AND
_Raw >= 0
THEN
PROC_AMP_Pyre_Write(_CombatID, _Raw);

PROC
PROC_AMP_Pyre_Commit((GUIDSTRING)_CombatID, (INTEGER)_Raw)
AND
_Raw < 0
THEN
PROC_AMP_Pyre_Write(_CombatID, 0);

PROC
PROC_AMP_Pyre_Write((GUIDSTRING)_CombatID, (INTEGER)_N)
AND
DB_AMP_Pyre_Points(_CombatID, (INTEGER)_Old)
THEN
NOT DB_AMP_Pyre_Points(_CombatID, _Old);
DB_AMP_Pyre_Points(_CombatID, _N);

PROC
PROC_AMP_Pyre_Write((GUIDSTRING)_CombatID, (INTEGER)_N)
AND
NOT DB_AMP_Pyre_Points(_CombatID, _)
THEN
DB_AMP_Pyre_Points(_CombatID, _N);

// effective points for grading: party-level ramp first (slower at low levels,
// full by ~10), then the hard multiplier, then thresholds 8/18/30/42/54/66/78/100/135/180.
// stored points untouched, this only picks the gradation.
PROC
PROC_AMP_Pyre_Recompute((GUIDSTRING)_CombatID, (INTEGER)_N)
AND
DB_AMP_AvgPartyLevel((INTEGER)_L)
THEN
PROC_AMP_Pyre_ScaleByLevel(_CombatID, _N, _L);

// party level unknown yet: treat as full speed
PROC
PROC_AMP_Pyre_Recompute((GUIDSTRING)_CombatID, (INTEGER)_N)
AND
NOT DB_AMP_AvgPartyLevel(_)
THEN
PROC_AMP_Pyre_GradeFinal(_CombatID, _N);

// level ramp: <=3 x2/5, 4-6 x3/5, 7-9 x4/5, 10+ full
PROC
PROC_AMP_Pyre_ScaleByLevel((GUIDSTRING)_CombatID, (INTEGER)_N, (INTEGER)_L)
AND
_L <= 3
AND
IntegerProduct(_N, 2, (INTEGER)_X)
AND
IntegerDivide(_X, 5, (INTEGER)_S)
THEN
PROC_AMP_Pyre_GradeFinal(_CombatID, _S);

PROC
PROC_AMP_Pyre_ScaleByLevel((GUIDSTRING)_CombatID, (INTEGER)_N, (INTEGER)_L)
AND
_L >= 4
AND
_L <= 6
AND
IntegerProduct(_N, 3, (INTEGER)_X)
AND
IntegerDivide(_X, 5, (INTEGER)_S)
THEN
PROC_AMP_Pyre_GradeFinal(_CombatID, _S);

PROC
PROC_AMP_Pyre_ScaleByLevel((GUIDSTRING)_CombatID, (INTEGER)_N, (INTEGER)_L)
AND
_L >= 7
AND
_L <= 9
AND
IntegerProduct(_N, 4, (INTEGER)_X)
AND
IntegerDivide(_X, 5, (INTEGER)_S)
THEN
PROC_AMP_Pyre_GradeFinal(_CombatID, _S);

PROC
PROC_AMP_Pyre_ScaleByLevel((GUIDSTRING)_CombatID, (INTEGER)_N, (INTEGER)_L)
AND
_L >= 10
THEN
PROC_AMP_Pyre_GradeFinal(_CombatID, _N);

// hard grades 25% hotter, on top of the level ramp
PROC
PROC_AMP_Pyre_GradeFinal((GUIDSTRING)_CombatID, (INTEGER)_S)
AND
DB_AMP_Difficulty_Mode("IMPOSSIBLE")

PROC_AMP_Pyre_GradeFinal((GUIDSTRING)_CombatID, (INTEGER)_S)
