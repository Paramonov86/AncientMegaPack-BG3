// sell execution
PROC
PROC_AMP_AS_SellItem((ITEM)_Item, (CHARACTER)_Player, (INTEGER)_SellPrice)
AND
_SellPrice >= 1
THEN
DebugText(_Player, "[AS] SOLD for gold");
AddGoldToMagicPockets(_Player, _SellPrice);
ObjectTimerLaunch(_Item, "AMP_AS_DeferDelete", 100, 0);
PlaySound(_Player, "UI_Gen_Coins_01");

PROC
PROC_AMP_AS_SellItem((ITEM)_Item, (CHARACTER)_Player, (INTEGER)_SellPrice)
AND
_SellPrice < 1
THEN
DebugText(_Player, "[AS] Sell for minimum 1g");
AddGoldToMagicPockets(_Player, 1);
ObjectTimerLaunch(_Item, "AMP_AS_DeferDelete", 100, 0);
PlaySound(_Player, "UI_Gen_Coins_01");

// deferred delete: deleting a just-picked-up item in the AddedTo frame
// null-derefs Wwise for certain templates (milk bottle, barrel). delete one
// tick later so the engine has finished the item's pickup audio setup.
IF
ObjectTimerFinished((ITEM)_Item, "AMP_AS_DeferDelete")
THEN
RequestDelete(_Item);
