SubGoalCombiner SGC_AND
INITSECTION
// maps caster equip-tier marker -> ally marker applied when the wearer puts any ilmater
// stigma on that ally. "your stigma" = any stigma cast by the wearer, not the one from this
// specific artifact (so diadem providence fires on an ally carrying the wearer's amulet stigma too).

// diadem providence: ally under any stigma you cast regens on turn start while wounded in combat. scales by diadem tier.
DB_AMP_Ilmater_PassiveMarker("AMP_ILMATER_DIADEM_T2_EQUIPPED", "AMP_ILMATER_PROVIDENCE_REGEN_T2");
DB_AMP_Ilmater_PassiveMarker("AMP_ILMATER_DIADEM_T3_EQUIPPED", "AMP_ILMATER_PROVIDENCE_REGEN_T3");
DB_AMP_Ilmater_PassiveMarker("AMP_ILMATER_DIADEM_T4_EQUIPPED", "AMP_ILMATER_PROVIDENCE_REGEN_T4");

// diadem retribution (t4 only): ally under any stigma you cast retaliates with a divine smite when hit in melee.
DB_AMP_Ilmater_PassiveMarker("AMP_ILMATER_DIADEM_T4_EQUIPPED", "AMP_ILMATER_RETRIBUTION_MARKER_T4");

// boots prophet: ally under any stigma you cast rolls for an inspiration point when hit with a status in combat. chance scales by tier.
DB_AMP_Ilmater_PassiveMarker("AMP_ILMATER_BOOTS_T1_EQUIPPED", "AMP_ILMATER_PROPHET_MARKER_T1");
DB_AMP_Ilmater_PassiveMarker("AMP_ILMATER_BOOTS_T2_EQUIPPED", "AMP_ILMATER_PROPHET_MARKER_T2");
DB_AMP_Ilmater_PassiveMarker("AMP_ILMATER_BOOTS_T3_EQUIPPED", "AMP_ILMATER_PROPHET_MARKER_T3");
DB_AMP_Ilmater_PassiveMarker("AMP_ILMATER_BOOTS_T4_EQUIPPED", "AMP_ILMATER_PROPHET_MARKER_T4");

KBSECTION

IF
StatusApplied((CHARACTER)_Target, _Stigma, (CHARACTER)_Caster, _)
AND
DB_AMP_Ilmater_StigmaType(_Stigma)
AND
DB_AMP_Ilmater_PassiveMarker(_TierMarker, _RegenMarker)
AND
HasActiveStatus(_Caster, _TierMarker, _HasTier)
AND
_HasTier == 1
THEN
ApplyStatus(_Target, _RegenMarker, -1.0, 1, _Caster);

EXITSECTION
ENDEXITSECTION
