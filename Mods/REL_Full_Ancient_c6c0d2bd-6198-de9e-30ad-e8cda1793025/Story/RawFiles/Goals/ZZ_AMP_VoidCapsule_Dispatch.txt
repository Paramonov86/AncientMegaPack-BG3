// dispatch - addedto handler with unified busy serialization
// routes coin/reroll/shard/artifact to respective start procs.
// perf: a single cheap gate (player-owned container) runs qry_..._iscontainer
//   once per addedto, then routes via proc_amp_voidcapsule_onitemadded.
//   previously 8 rules each ran iscontainer (gettemplate) on every addedto in
//   the game (loot/equip/combat) -> ~10k gettemplate calls -> combat fps drops.
//   logic is unchanged; only the shared head (owner+player+iscontainer) is hoisted.

// gate: only items added to a player-owned voidcapsule reach the router
// cheap filters first: getinventoryowner fails fast on world/npc containers,
// so iscontainer (gettemplate) never runs for them.
IF
AddedTo((ITEM)_Item, (ITEM)_Capsule, _)
AND
GetInventoryOwner(_Capsule, (CHARACTER)_Actor)
AND
DB_Players(_Actor)
AND
QRY_AMP_VoidCapsule_IsContainer(_Capsule)
THEN
PROC_AMP_VoidCapsule_OnItemAdded(_Item, _Capsule, _Actor);

// rule 1: coin success
PROC
PROC_AMP_VoidCapsule_OnItemAdded((ITEM)_Coin, (ITEM)_Capsule, (CHARACTER)_Actor)
AND
QRY_AMP_VoidCapsule_IsCoin(_Coin)
AND
HasActiveStatus(_Coin, "TOT_SHOP_ITEM", 0)
AND
NOT DB_AMP_VoidCapsule_CoinProcessed(_Coin)
AND
NOT DB_AMP_VoidCapsule_Depleted(_Capsule)
AND
NOT DB_AMP_VoidCapsule_Busy(_Capsule)
THEN
DB_AMP_VoidCapsule_CoinProcessed(_Coin);
DB_AMP_VoidCapsule_Busy(_Capsule);
ObjectTimerLaunch(_Capsule, "AMP_VoidCapsule_BusySafety", 2000, 0);
PROC_AMP_VoidCapsule_SetActionUser(_Capsule, _Actor);
PROC_AMP_VoidCapsule_CoinStart(_Coin, _Capsule, _Actor);

// rule 2a: reroll on fresh capsule (no charges yet, max >= cost)
PROC
PROC_AMP_VoidCapsule_OnItemAdded((ITEM)_Item, (ITEM)_Capsule, (CHARACTER)_Actor)
AND
QRY_AMP_VoidCapsule_IsRerollItem(_Item)
AND
HasActiveStatus(_Item, "TOT_SHOP_ITEM", 0)
AND
NOT DB_AMP_VoidCapsule_RerollProcessing(_Item)
AND
NOT DB_AMP_VoidCapsule_Depleted(_Capsule)
AND
NOT DB_AMP_VoidCapsule_Busy(_Capsule)
AND
NOT DB_AMP_VoidCapsule_Charges(_Capsule, _)
AND
DB_AMP_VoidCapsule_ChargesMaxActive((INTEGER)_Max)
AND
DB_AMP_VoidCapsule_RerollCostActive((INTEGER)_Cost)
AND
_Max >= _Cost
THEN
DB_AMP_VoidCapsule_RerollProcessing(_Item);
DB_AMP_VoidCapsule_Busy(_Capsule);
ObjectTimerLaunch(_Capsule, "AMP_VoidCapsule_BusySafety", 2000, 0);
PROC_AMP_VoidCapsule_SetActionUser(_Capsule, _Actor);
PROC_AMP_VoidCapsule_RerollStart(_Item, _Capsule, _Actor);

// rule 2b: reroll on existing capsule (charges >= cost)
PROC
PROC_AMP_VoidCapsule_OnItemAdded((ITEM)_Item, (ITEM)_Capsule, (CHARACTER)_Actor)
AND
QRY_AMP_VoidCapsule_IsRerollItem(_Item)
AND
HasActiveStatus(_Item, "TOT_SHOP_ITEM", 0)
AND
NOT DB_AMP_VoidCapsule_RerollProcessing(_Item)
AND
NOT DB_AMP_VoidCapsule_Depleted(_Capsule)
AND
NOT DB_AMP_VoidCapsule_Busy(_Capsule)
AND
DB_AMP_VoidCapsule_Charges(_Capsule, (INTEGER)_Charges)
AND
DB_AMP_VoidCapsule_RerollCostActive((INTEGER)_Cost)
AND
_Charges >= _Cost
THEN
DB_AMP_VoidCapsule_RerollProcessing(_Item);
DB_AMP_VoidCapsule_Busy(_Capsule);
ObjectTimerLaunch(_Capsule, "AMP_VoidCapsule_BusySafety", 2000, 0);
PROC_AMP_VoidCapsule_SetActionUser(_Capsule, _Actor);
PROC_AMP_VoidCapsule_RerollStart(_Item, _Capsule, _Actor);

// rule 2c: reroll with insufficient charges (eject, no resource cost)
PROC
PROC_AMP_VoidCapsule_OnItemAdded((ITEM)_Item, (ITEM)_Capsule, (CHARACTER)_Actor)
AND
QRY_AMP_VoidCapsule_IsRerollItem(_Item)
AND
HasActiveStatus(_Item, "TOT_SHOP_ITEM", 0)
AND
NOT DB_AMP_VoidCapsule_Depleted(_Capsule)
AND
NOT DB_AMP_VoidCapsule_Busy(_Capsule)
AND
DB_AMP_VoidCapsule_Charges(_Capsule, (INTEGER)_Charges)
AND
DB_AMP_VoidCapsule_RerollCostActive((INTEGER)_Cost)
AND
_Charges < _Cost
THEN
PlaySound(_Actor, "UI_HUD_Combine_Error");
ObjectTimerLaunch(_Item, "AMP_VoidCapsule_Eject", 100, 0);
