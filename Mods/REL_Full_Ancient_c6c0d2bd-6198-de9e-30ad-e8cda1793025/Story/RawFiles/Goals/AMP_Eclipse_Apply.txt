// apply: typed info if mapped, else base radiant.
// classify writes a scratch fact, both resolve rules read it, then it's wiped.
PROC
PROC_AMP_Eclipse_ApplyInfo((CHARACTER)_C)
THEN
PROC_AMP_Eclipse_Classify(_C);
PROC_AMP_Eclipse_ApplyInfo_Resolve(_C);
PROC_AMP_Eclipse_ClearResolved(_C);

// mapped boss: clear stale base carrier then apply the typed one
PROC
PROC_AMP_Eclipse_ApplyInfo_Resolve((CHARACTER)_C)
AND
DB_AMP_Eclipse_Resolved(_C, (STRING)_Info)
THEN
RemoveStatus(_C, "AMP_ECLIPSE_INFO", NULL_00000000-0000-0000-0000-000000000000);
ApplyStatus(_C, _Info, -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);
DB_AMP_Eclipse_HasInfo(_C);

// unmapped boss (incl. mod bosses): default radiant carrier
PROC
PROC_AMP_Eclipse_ApplyInfo_Resolve((CHARACTER)_C)
AND
NOT DB_AMP_Eclipse_Resolved(_C, _)
THEN
ApplyStatus(_C, "AMP_ECLIPSE_INFO", -1.0, 1, NULL_00000000-0000-0000-0000-000000000000);
DB_AMP_Eclipse_HasInfo(_C);

// wipe the scratch fact
PROC
PROC_AMP_Eclipse_ClearResolved((CHARACTER)_C)
AND
DB_AMP_Eclipse_Resolved(_C, (STRING)_X)
THEN
NOT DB_AMP_Eclipse_Resolved(_C, _X);

// remove: strip both base and typed carriers
PROC
PROC_AMP_Eclipse_RemoveInfo((CHARACTER)_C)
THEN
RemoveStatus(_C, "AMP_ECLIPSE_INFO", NULL_00000000-0000-0000-0000-000000000000);
PROC_AMP_Eclipse_Classify(_C);
PROC_AMP_Eclipse_RemoveInfo_Resolve(_C);
PROC_AMP_Eclipse_ClearResolved(_C);
NOT DB_AMP_Eclipse_HasInfo(_C);

PROC
PROC_AMP_Eclipse_RemoveInfo_Resolve((CHARACTER)_C)
AND
DB_AMP_Eclipse_Resolved(_C, (STRING)_Info)
THEN
RemoveStatus(_C, _Info, NULL_00000000-0000-0000-0000-000000000000);

// activate on switch to hard

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "HARD")
AND
NOT DB_AMP_Eclipse_Active(1)
THEN
DB_AMP_Eclipse_Active(1);
PROC_AMP_Eclipse_ApplyToAllBosses();

PROC
PROC_AMP_Eclipse_ApplyToAllBosses()
AND
DB_AMP_DiffV3_Tracked((CHARACTER)_C, (STRING)_S)
AND
IsBoss(_C, 1)
AND
IsPlayer(_C, 0)
AND
NOT DB_AMP_Eclipse_HasInfo(_C)
THEN
PROC_AMP_Eclipse_ApplyInfo(_C);

// deactivate when leaving hard for any other mode

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "CASUAL")
AND
DB_AMP_Eclipse_Active(1)
THEN
NOT DB_AMP_Eclipse_Active(1);
PROC_AMP_Eclipse_CleanupAllBosses();

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "NORMAL")
AND
DB_AMP_Eclipse_Active(1)
THEN
NOT DB_AMP_Eclipse_Active(1);
PROC_AMP_Eclipse_CleanupAllBosses();

PROC
PROC_AMP_SetGameMode((CHARACTER)_Host, "STOCK")
AND
DB_AMP_Eclipse_Active(1)
THEN
NOT DB_AMP_Eclipse_Active(1);
PROC_AMP_Eclipse_CleanupAllBosses();

PROC
PROC_AMP_Eclipse_CleanupAllBosses()
AND
DB_AMP_Eclipse_HasInfo((CHARACTER)_C)
THEN
PROC_AMP_Eclipse_RemoveInfo(_C);

// auto-apply to a newly tracked boss while active

IF

AND
DB_AMP_Eclipse_Active(1)
AND
IsBoss(_C, 1)
AND
IsPlayer(_C, 0)
AND
NOT DB_AMP_Eclipse_HasInfo(_C)

PROC_AMP_Eclipse_ApplyInfo(_C);

// strip on recruit (e.g. minthara joins mid-game)

IF

AND
DB_AMP_Eclipse_HasInfo(_C)
THEN
PROC_AMP_Eclipse_RemoveInfo(_C);

// form change wipes the carrier but the HasInfo marker survives in the db.
// clear it and re-apply so transforming bosses (orin slayer etc) keep eclipse.

IF

AND
DB_AMP_Eclipse_Active(1)
AND
IsBoss(_C, 1)
AND
IsPlayer(_C, 0)
AND
DB_AMP_Eclipse_HasInfo(_C)
THEN
NOT DB_AMP_Eclipse_HasInfo(_C);
PROC_AMP_Eclipse_ApplyInfo(_C);

IF
ShapeshiftChanged((CHARACTER)_C, _, _, _)

DB_AMP_Eclipse_Active(1)
AND

AND
IsPlayer(_C, 0)
AND
DB_AMP_Eclipse_HasInfo(_C)
THEN
NOT DB_AMP_Eclipse_HasInfo(_C);
PROC_AMP_Eclipse_ApplyInfo(_C);

// bootstrap: activate on level load if hard was already set

IF
LevelGameplayStarted(_, _)

DB_AMP_GameMode("HARD")
AND
NOT DB_AMP_Eclipse_Active(1)

NOT DB_AMP_Eclipse_TypedMigrated(1)
THEN
