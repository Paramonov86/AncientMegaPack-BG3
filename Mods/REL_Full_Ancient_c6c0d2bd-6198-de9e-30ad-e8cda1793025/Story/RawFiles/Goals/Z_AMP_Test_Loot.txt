KBSECTION
// spell casting - spawn chests
IF
CastedSpell((CHARACTER)_Character, "Shout_AMP_Test", _, _, _)
AND
IsPlayer(_Character, 1)
THEN
PROC_AMP_Para_SpawnAllChests(_Character);

PROC
PROC_AMP_Para_SpawnAllChests((CHARACTER)_Character)
AND
DB_AMP_Para_Template(_Template)
THEN
TemplateAddTo(_Template, _Character, 1, 0);

// final effect after all chests spawned
IF
CastedSpell((CHARACTER)_Character, "Shout_AMP_Test", _, _, _)
AND
IsPlayer(_Character, 1)
THEN
PROC_AMP_PlayModeEffect(_Character);

// track spawned chests
IF
TemplateAddedTo((ITEMROOT)_Template, (ITEM)_Chest, (CHARACTER)_Owner, _)
AND
DB_AMP_Para_Template(_Template)
THEN
DB_AMP_Para_Spawned(_Chest);

// auto-delete when dropped
IF
RemovedFrom((ITEM)_Chest, (GUIDSTRING)_Container)
AND
DB_AMP_Para_Spawned(_Chest)
THEN
PROC_AMP_Para_CheckAndDelete(_Chest);

PROC
PROC_AMP_Para_CheckAndDelete((ITEM)_Chest)
AND
NOT QRY_AMP_Para_IsInAnyPlayerInventory(_Chest)
AND
GetPosition(_Chest, _X, _Y, _Z)
AND
RealSubtract(_Y, 1.0, _YLow)
THEN
PROC_AMP_PlayModeEffectAtPos(_X, _YLow, _Z);
ObjectTimerLaunch(_Chest, "AMP_Para_Delete", 500, 0);

QRY
QRY_AMP_Para_IsInAnyPlayerInventory((ITEM)_Chest)
AND
DB_Players(_Player)
AND
IsInInventoryOf(_Chest, _Player, 1)
THEN
DB_NOOP(1);

// deletion timer
IF
ObjectTimerFinished((ITEM)_Chest, "AMP_Para_Delete")
AND
DB_AMP_Para_Spawned(_Chest)
AND
Exists(_Chest, 1)
THEN
NOT DB_AMP_Para_Spawned(_Chest);
RequestDelete(_Chest);

// cleanup on destroy
IF
DestroyedBy((ITEM)_Chest, _, _, _)
AND
DB_AMP_Para_Spawned(_Chest)
THEN
NOT DB_AMP_Para_Spawned(_Chest);
