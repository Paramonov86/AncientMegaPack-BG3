//region special redirect intercept (quest-locked traders like brem)
// when vanilla sets a table that has a special redirect, use redirect target instead

PROC
PROC_SetCustomTradeTreasure((CHARACTER)_Trader, (STRING)_VanillaTable)
AND
DB_AMP_TradingFix_Enabled(1)
AND
DB_AMP_SpecialRedirect(_Trader, _VanillaTable, _RedirectAMP)
AND
DB_AMP_CurrentTable(_Trader, _OldAMP)
AND
_OldAMP != _RedirectAMP
THEN
NOT DB_AMP_CurrentTable(_Trader, _OldAMP);
DB_AMP_CurrentTable(_Trader, _RedirectAMP);
NOT DB_AMP_Visited(_Trader);

//end_region

//region intercept proc_setcustomtradetreasure (generic)
// when vanilla changes table, we update our tracking and reset visited
// skip if specialredirect already handled this trader+table combo

PROC
PROC_SetCustomTradeTreasure((CHARACTER)_Trader, (STRING)_VanillaTable)
AND
DB_AMP_TradingFix_Enabled(1)
AND
NOT DB_AMP_SpecialRedirect(_Trader, _VanillaTable, _)
AND
DB_AMP_CurrentTable(_Trader, _OldAMP)
AND
Concatenate(_VanillaTable, "_AMP", _NewAMP)
AND
_OldAMP != _NewAMP
THEN
NOT DB_AMP_CurrentTable(_Trader, _OldAMP);
DB_AMP_CurrentTable(_Trader, _NewAMP);
NOT DB_AMP_Visited(_Trader);

//end_region

//region thrumbo special case
// thrumbo is not a trader until beggartakeover event
// vanilla uses direct setcustomtradetreasure (not proc_setcustomtradetreasure)
// after event he inherits mystic carrion's table

PROC
PROC_LOW_PhilgravesMansion_BeggarTakeover()

THEN

//end_region

//region requesttrade handling

// first visit -> _amp table
IF
RequestTrade((CHARACTER)_Player, (CHARACTER)_Trader, _, _)
AND
DB_AMP_TradingFix_Enabled(1)
AND
DB_AMP_CurrentTable(_Trader, _AMPTable)
AND
NOT DB_AMP_Visited(_Trader)
THEN
SetCustomTradeTreasure(_Trader, _AMPTable);

// repeat visit -> _none table
IF
RequestTrade((CHARACTER)_Player, (CHARACTER)_Trader, _, _)
AND
DB_AMP_TradingFix_Enabled(1)
AND
DB_AMP_CurrentTable(_Trader, _AMPTable)
AND
DB_AMP_ToNone(_AMPTable, _NoneTable)
AND
DB_AMP_Visited(_Trader)
THEN
SetCustomTradeTreasure(_Trader, _NoneTable);

//end_region

//region requestpickpocket handling

DB_AMP_TradingFix_Enabled(1)
AND
DB_AMP_CurrentTable(_Trader, _AMPTable)
AND
NOT DB_AMP_Visited(_Trader)
THEN

// repeat visit -> _none table
IF
RequestPickpocket((CHARACTER)_Player, (CHARACTER)_Trader)
AND
DB_AMP_TradingFix_Enabled(1)
AND
DB_AMP_CurrentTable(_Trader, _AMPTable)
AND
DB_AMP_ToNone(_AMPTable, _NoneTable)
AND
DB_AMP_Visited(_Trader)
THEN
SetCustomTradeTreasure(_Trader, _NoneTable);

//end_region

//region mark visited
// mark as visited immediately when player opens trade/pickpocket window
// player already sees the items - no point waiting for transaction result

// trade opened (first visit)
IF
RequestTrade((CHARACTER)_Player, (CHARACTER)_Trader, _, _)
AND
DB_AMP_TradingFix_Enabled(1)
AND
DB_AMP_CurrentTable(_Trader, _)
AND
NOT DB_AMP_Visited(_Trader)
THEN

// pickpocket opened (first visit)
IF
RequestPickpocket((CHARACTER)_Player, (CHARACTER)_Trader)
AND
DB_AMP_TradingFix_Enabled(1)
AND
DB_AMP_CurrentTable(_Trader, _)
AND
NOT DB_AMP_Visited(_Trader)
THEN
DB_AMP_Visited(_Trader);

//end_region
