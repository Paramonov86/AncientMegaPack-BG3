// rumors: normal -3 per same-act trader
PROC
PROC_AMP_SpreadRumors((CHARACTER)_Thief, (CHARACTER)_Victim)
AND
DB_AMP_TraderAct(_Victim, (INTEGER)_Act)
AND
DB_AMP_TraderAct((CHARACTER)_OtherTrader, _Act)
AND
_OtherTrader != _Victim
AND
DB_AMP_Difficulty_Mode("STANDARD")
AND
QRY_AMP_AntiTheft_Active()
THEN
AddAttitudeTowardsPlayer(_OtherTrader, _Thief, -3);

// rumors: hard -5 per same-act trader
PROC
PROC_AMP_SpreadRumors((CHARACTER)_Thief, (CHARACTER)_Victim)
AND
DB_AMP_TraderAct(_Victim, (INTEGER)_Act)
AND
DB_AMP_TraderAct((CHARACTER)_OtherTrader, _Act)
AND
_OtherTrader != _Victim
AND
DB_AMP_Difficulty_Mode("IMPOSSIBLE")
AND
QRY_AMP_AntiTheft_Active()
THEN
AddAttitudeTowardsPlayer(_OtherTrader, _Thief, -5);

// hub trader defense: gold-delta tracking + fail gold wipe + delayed perma hostility.
// 20k respawn block lives in AMP_TraderGold_x100.txt (driven by DB_AMP_HubEverStolen).

// turn on vanilla EmptyPocketNoticed + Theft crime for hub traders at level load
IF
LevelGameplayStarted(_, _)
AND
DB_AMP_HubTrader((CHARACTER)_Npc)
AND
Exists(_Npc, 1)
AND
QRY_AMP_AntiTheft_Active()
THEN
CharacterEnableCrime(_Npc, "EmptyPocketNoticed");
CharacterEnableCrime(_Npc, "Theft");

// projections / holograms - physically unpickpocketable
IF
LevelGameplayStarted(_, _)
AND
DB_AMP_HubProjection((CHARACTER)_Projection)
AND
Exists(_Projection, 1)
THEN
SetCanBePickpocketed(_Projection, 0);

// snapshot player gold at roll-request so we can measure the delta after a success.
// refresh on each attempt so the delta tracks the current roll.

IF
RequestPickpocket((CHARACTER)_Player, (CHARACTER)_NPC)
AND
DB_AMP_HubTrader(_NPC)
AND
GetGold(_Player, (INTEGER)_PreGold)
AND
NOT DB_AMP_PrePickpocketGold(_Player, _NPC, _)
THEN
DB_AMP_PrePickpocketGold(_Player, _NPC, _PreGold);

IF
RequestPickpocket((CHARACTER)_Player, (CHARACTER)_NPC)
AND
DB_AMP_HubTrader(_NPC)
AND
GetGold(_Player, (INTEGER)_PreGold)
AND
DB_AMP_PrePickpocketGold(_Player, _NPC, (INTEGER)_OldPre)
THEN
NOT DB_AMP_PrePickpocketGold(_Player, _NPC, _OldPre);
DB_AMP_PrePickpocketGold(_Player, _NPC, _PreGold);

// success at hub: mark ever-stolen (blocks the 20k respawn), and if gold delta >= 500
// schedule perma hostility.

IF
CharacterPickpocketSuccess((CHARACTER)_Player, (CHARACTER)_NPC, (ITEM)_Item, (GUIDSTRING)_Tpl, (INTEGER)_Amount, (INTEGER)_GoldValue)
AND
DB_AMP_HubTrader(_NPC)
AND
QRY_AMP_AntiTheft_Active()
AND
NOT DB_AMP_HubEverStolen(_NPC, _Player)
THEN
DB_AMP_HubEverStolen(_NPC, _Player);

IF
CharacterPickpocketSuccess((CHARACTER)_Player, (CHARACTER)_NPC, (ITEM)_Item, (GUIDSTRING)_Tpl, (INTEGER)_Amount, (INTEGER)_GoldValue)
AND
DB_AMP_HubTrader(_NPC)
AND
QRY_AMP_AntiTheft_Active()
AND
DB_AMP_PrePickpocketGold(_Player, _NPC, (INTEGER)_PreGold)
AND
GetGold(_Player, (INTEGER)_NowGold)
AND
IntegerSubtract(_NowGold, _PreGold, (INTEGER)_Delta)
AND
_Delta >= 500
AND
DB_AMP_HubFaction(_NPC, (FACTION)_Fac)
AND
NOT DB_AMP_HostilityPending(_NPC, _, _)
THEN
DB_AMP_HostilityPending(_NPC, _Player, _Fac);
RealtimeObjectTimerLaunch(_NPC, "AMP_HubPerma", 120000);

// fail at hub: strip invis so vanilla crime can see the culprit, register Theft,
// wipe the trader's gold (hides the stash), mark ever-stolen, start the 120s perma timer.

IF
CharacterPickpocketFailed((CHARACTER)_Player, (CHARACTER)_NPC)
AND
DB_AMP_HubTrader(_NPC)
AND
QRY_AMP_AntiTheft_Active()
AND
GetPosition(_NPC, (REAL)_X, (REAL)_Y, (REAL)_Z)
AND
CrimeGetNewID((INTEGER)_CrimeID)
THEN
RemoveStatusesWithGroup(_Player, "SG_Invisible", _NPC);
PROC_CharacterRegisterCrimeWithPosition(_Player, "Theft", NULL_00000000-0000-0000-0000-000000000000, _X, _Y, _Z, _NPC, _CrimeID);
PROC_AMP_HubFailAftermath(_Player, _NPC);

PROC
PROC_AMP_HubFailAftermath((CHARACTER)_Player, (CHARACTER)_NPC)
AND
GetGold(_NPC, (INTEGER)_NpcGold)
AND
_NpcGold > 0
AND
IntegerSubtract(0, _NpcGold, (INTEGER)_Neg)
THEN
AddGold(_NPC, _Neg);

PROC
PROC_AMP_HubFailAftermath((CHARACTER)_Player, (CHARACTER)_NPC)
AND
NOT DB_AMP_HubEverStolen(_NPC, _Player)
THEN
DB_AMP_HubEverStolen(_NPC, _Player);

PROC
PROC_AMP_HubFailAftermath((CHARACTER)_Player, (CHARACTER)_NPC)
AND
DB_AMP_HubFaction(_NPC, (FACTION)_Fac)
AND
NOT DB_AMP_HostilityPending(_NPC, _, _)
THEN
DB_AMP_HostilityPending(_NPC, _Player, _Fac);
RealtimeObjectTimerLaunch(_NPC, "AMP_HubPerma", 120000);

// murder of hub trader: force the Murder crime. vanilla needs a witness, so an
// ambush with nobody nearby would otherwise skip it.

IF
Died((CHARACTER)_NPC)
AND
DB_AMP_HubTrader(_NPC)
AND
QRY_AMP_AntiTheft_Active()
AND
GetHostCharacter((CHARACTER)_Host)
AND
CrimeGetNewID((INTEGER)_CrimeID)
THEN
PROC_CharacterRegisterCrime(_Host, "Murder", NULL_00000000-0000-0000-0000-000000000000, _NPC, _CrimeID);

// perma hostility on timer fire, with a level-load backup. the realtime timer can
// die on level unload / npc despawn, so the backup applies pending entries on load
// and porting away can't dodge it.

IF
ObjectTimerFinished((GUIDSTRING)_NPC, "AMP_HubPerma")
AND
DB_AMP_HostilityPending((CHARACTER)_NPC, (CHARACTER)_Player, (FACTION)_Fac)
THEN
PROC_AMP_HubApplyPerma(_NPC, _Player, _Fac);

IF
LevelGameplayStarted(_, _)
AND
DB_AMP_HostilityPending((CHARACTER)_NPC, (CHARACTER)_Player, (FACTION)_Fac)
THEN
PROC_AMP_HubApplyPerma(_NPC, _Player, _Fac);

// apply perma: persistent faction hostility to every player faction, plus immediate
// aggro if the npc is on-stage.

PROC
PROC_AMP_HubApplyPerma((CHARACTER)_NPC, (CHARACTER)_Player, (FACTION)_Fac)
AND
DB_Players((CHARACTER)_P)
AND
GetFaction(_P, (FACTION)_PF)
AND
IntegerSubtract(0, 100, (INTEGER)_Neg100)
THEN
SetRelation(_Fac, _PF, _Neg100);
DB_HostileToPlayerGroup(_Fac, _P);

PROC
PROC_AMP_HubApplyPerma((CHARACTER)_NPC, (CHARACTER)_Player, (FACTION)_Fac)
AND
IsOnStage(_NPC, 1)
AND
GetFaction(_Player, (FACTION)_PF)
THEN
SetHostileAndEnterCombat(_Fac, _PF, _NPC, _Player);

PROC
PROC_AMP_HubApplyPerma((CHARACTER)_NPC, _, _)
AND
DB_AMP_HostilityPending(_NPC, (CHARACTER)_Player, (FACTION)_Fac)
THEN
NOT DB_AMP_HostilityPending(_NPC, _Player, _Fac);
